MODEL = "muse-spark-1.3"

def _make_body(content: list, max_tokens: int = 4000, retry: bool = False) -> dict:
    body = {"model": MODEL, "input": [user_message(content)], "reasoning": {"effort": "low"},
            "max_output_tokens": 8000 if retry else max_tokens}
    if retry:
        # Extra field changes the request-json hash, so a cached (possibly
        # empty) first response is not returned again from the disk cache.
        body["metadata"] = {"retry": "1"}
    return body

def _call(content: list, max_tokens: int = 4000, retry: bool = False) -> str:
    return extract_text(responses(_make_body(content, max_tokens=max_tokens, retry=retry)))

def _call_json(content: list, max_tokens: int = 4000, what: str = "spark"):
    """Call Spark and parse strict JSON, retrying once on empty/bad output.

    Retry uses metadata={'retry': '1'} (bypasses the disk cache for the
    original request), max_output_tokens=8000 and reasoning effort 'low'.
    Raises the last parse/empty error if both attempts fail, so callers can
    return their conservative defaults. Empty responses are never treated
    as valid and are not (re-)served for the retry thanks to the changed
    request body / cache key.
    """
    last_exc: Exception | None = None
    for attempt in (0, 1):
        retry = attempt == 1
        text = _call(content, max_tokens=max_tokens, retry=retry)
        if not text or not str(text).strip():
            last_exc = ValueError("empty text, cannot parse JSON")
            log.warning("%s: empty text from Spark (attempt %d/2)%s",
                        what, attempt + 1, "; retrying with larger budget" if not retry else "")
            continue
        try:
            return parse_json_loose(text)
        except Exception as e:
            last_exc = e
            log.warning("%s parse failed (attempt %d/2): %s; text=%s",
                        what, attempt + 1, e, str(text)[:500])
            continue
    assert last_exc is not None
    raise last_exc

def list_objects(before_frames: list, after_frames: list, concepts=None) -> list[dict]:
    """Inventory objects from keyframes of both videos, or honour override."""
    if concepts:

        return [{"concept": c, "where_before": "", "where_after": "", "suspected_change": "unchanged"} for c in concepts]
    content = [text_part(
        "You are inventorying a room from phone-video keyframes. Images are labelled BEFORE frame i and AFTER frame i; "
        "camera viewpoint may differ. List at most 12 movable household objects visible in either set. Prioritise movable "
        "household objects (lamp, chair, backpack, suitcase, walker, plant, box, book, cushion, etc.), short singular noun "
        "phrases SAM can segment. Return STRICT JSON only, no markdown: "
        '{"objects": [{"concept": "floor lamp", "where_before": "str", "where_after": "str", '
        '"suspected_change": "added|removed|moved|unchanged"}]}')]
    for i, p in enumerate(before_frames):
        content.append(text_part(f"BEFORE frame {i}")); content.append(image_part(p))

        content.append(text_part(f"AFTER frame {i}")); content.append(image_part(p))
    try:
        data = _call_json(content, what="list_objects")
        objs = data.get("objects", []) if isinstance(data, dict) else data
    except Exception as e:
        log.warning("inventory parse failed: %s", e); objs = []
    if not isinstance(objs, (list, tuple)):
        objs = []
    clean = []
    for o in objs[:12]:
        if isinstance(o, dict) and o.get("concept"):

                          "where_before": str(o.get("where_before", "")), "where_after": str(o.get("where_after", "")),

    return clean

def _crop(frame_path, box, out_path):
    img = cv2.imread(str(frame_path))
    if img is None or not box:
        return None
    h, w = img.shape[:2]; x1, y1, x2, y2 = [int(v) for v in box]
    pad_x, pad_y = int(0.6 * max(1, x2 - x1)), int(0.6 * max(1, y2 - y1))
    crop = img[max(0, y1 - pad_y):min(h, y2 + pad_y), max(0, x1 - pad_x):min(w, x2 + pad_x)]
    if crop.size == 0:
        return None
    cv2.imwrite(str(out_path), crop); return str(out_path)

def judge_moved(concept: str, before_img, after_img, before_box=None, after_box=None) -> dict:

    tmp = Path("cache") / "judge"; tmp.mkdir(parents=True, exist_ok=True)

    bp, ap = tmp / f"{safe}_before.jpg", tmp / f"{safe}_after.jpg"
    cv2.imwrite(str(bp), draw_box(before_img, before_box, concept, (0, 165, 255)))
    cv2.imwrite(str(ap), draw_box(after_img, after_box, concept, (0, 165, 255)))
    content = [text_part(
        f"The orange box marks '{concept}' in the BEFORE frame and the AFTER frame of the same room. "
        "Camera viewpoint differs between videos, so do NOT judge by raw pixel position. Judge position relative to "
        "surrounding furniture and room landmarks. Also shown are context crops. Return STRICT JSON only: "
        '{"moved": true, "confidence": 0.0, "reason": "short reason"} with confidence in 0..1.'),
        text_part("BEFORE full frame"), image_part(str(bp)), text_part("AFTER full frame"), image_part(str(ap))]
    for label, src, box, name in (("BEFORE crop", str(bp), before_box, "b_crop"), ("AFTER crop", str(ap), after_box, "a_crop")):
        # crop from original annotated image area is approximate but gives context
        crop_path = _crop(src, box, tmp / f"{safe}_{name}.jpg")
        if crop_path:
            content.append(text_part(label)); content.append(image_part(crop_path))
    try:
        d = _call_json(content, max_tokens=1000, what=f"judge_moved {concept}")
        if not isinstance(d, dict):
            raise ValueError("empty text, cannot parse JSON" if not d else "parsed JSON is not an object")
        return {"moved": bool(d.get("moved")), "confidence": float(d.get("confidence", 0.0)), "reason": str(d.get("reason", ""))}
    except Exception as e:
        log.warning("judge_moved parse failed for %s: %s", concept, e)
        if "empty" in str(e).lower():
            return {"moved": False, "confidence": 0.0, "reason": "no response"}
        return {"moved": False, "confidence": 0.0, "reason": "unparseable Spark response"}

"""Pure deterministic diff + grounding filter. No API calls."""
from __future__ import annotations

PRESENCE_MIN_FRAMES = 2
CONF_THRESHOLD = 0.5
MOVED_MIN_CONF = 0.8

def summarize_track(parsed: dict) -> dict:
    """Summarise one concept in one video.

    ``best_frame``/``best_box`` are chosen by a demo-friendly score rather
    than raw confidence alone: detections cut off at the frame border are
    penalised (centrality 0.5 vs 1.0 for boxes at least 2 px inside every
    edge), with a small bonus for larger boxes:

        score = c * centrality * (0.75 + 0.25 * min(1, area_frac * 8))

    ``best_conf`` remains the raw confidence of the chosen detection, and
    ``mean_conf``/``frames_detected``/``max_instances`` are unchanged.
    """
    frames, confs, best_frame, best_box, best_c, best_score, max_inst = 0, [], None, None, -1.0, -1.0, 0
    for frame_idx in sorted((parsed or {}).keys(), key=lambda x: int(x)):
        dets = [d for d in parsed[frame_idx] if float(d.get("c", 0)) >= CONF_THRESHOLD]
        if not dets:
            continue
        frames += 1; max_inst = max(max_inst, len(dets))
        for d in dets:
            c = float(d.get("c", 0)); confs.append(c)
            box = list(d.get("box") or [])
            # Centrality / area need a valid box and frame size; be defensive.
            centrality, area_frac = 1.0, 0.0
            try:
                w, h = float(d.get("w", 0) or 0), float(d.get("h", 0) or 0)
                if len(box) == 4 and w > 0 and h > 0:
                    x1, y1, x2, y2 = (float(v) for v in box)
                    touches = x1 < 2 or y1 < 2 or x2 > w - 2 or y2 > h - 2
                    centrality = 0.5 if touches else 1.0
                    area_frac = max(0.0, (x2 - x1) * (y2 - y1)) / (w * h)
            except Exception:
                centrality, area_frac = 1.0, 0.0
            score = c * centrality * (0.75 + 0.25 * min(1.0, area_frac * 8))
            if score > best_score:
                best_score, best_c, best_frame, best_box = score, c, int(frame_idx), list(d.get("box"))
    return {"frames_detected": frames, "max_instances": max_inst,
            "mean_conf": (sum(confs) / len(confs)) if confs else 0.0,
            "best_frame": best_frame, "best_box": best_box,
            "best_conf": best_c if best_c >= 0 else 0.0, "present": frames >= PRESENCE_MIN_FRAMES}

def summarize_tracks(before_tracks: dict, after_tracks: dict) -> dict:
    """Per-concept summaries + SAM label for both videos."""
    out = {}
    for concept in sorted(set(before_tracks) | set(after_tracks)):
        b = summarize_track(before_tracks.get(concept, {}))
        a = summarize_track(after_tracks.get(concept, {}))
        if a["present"] and not b["present"]:
            label = "added"
        elif b["present"] and not a["present"]:
            label = "removed"
        elif b["present"] and a["present"]:
            label = "present_both"
        else:
            label = "not_found"
        out[concept] = {"concept": concept, "before": b, "after": a, "label": label,
                        "original_label": label, "recheck": None,
                        "count_changed": b["max_instances"] != a["max_instances"]}
    return out

def apply_image_recheck(summaries: dict, concept: str, recheck: dict | None) -> dict:
    """Record a SAM image re-check result in ``summaries`` (in place).

    ``recheck`` should contain at least ``video`` (``"before"``/``"after"``,
    the side where the object was supposedly absent), ``keyframe_index``,
    ``box`` and ``c`` when a detection was found. If any detection with
    c >= CONF_THRESHOLD was found, a video-tracking label of ``added`` /
    ``removed`` is corrected to ``present_both``. The original
    video-tracking label is preserved as ``original_label`` so callers can
    still show Spark (and the grounding filter) what video tracking said.
    Returns the updated summary (or {} if the concept is unknown).
    """
    s = summaries.get(concept)
    if s is None:
        return {}
    if "original_label" not in s or not s.get("original_label"):
        s["original_label"] = s.get("label")
    if recheck is None:
        s["recheck"] = {"checked": False, "found": False, "video": None,
                        "keyframe_index": None, "box": None, "c": 0.0}
        return s
    info = dict(recheck)
    info.setdefault("checked", True)
    # Normalise / derive the "found" flag from the confidence.
    try:
        c_val = float(info.get("c", 0.0) or 0.0)
    except Exception:
        c_val = 0.0
    info["c"] = c_val
    if info.get("box") is not None:
        try:
            info["box"] = list(info["box"])
        except Exception:
            pass
    if info.get("keyframe_index") is not None:
        try:
            info["keyframe_index"] = int(info["keyframe_index"])
        except Exception:
            pass
    info["found"] = bool(info.get("found", c_val >= CONF_THRESHOLD and info.get("box") is not None))
    s["recheck"] = info
    if info["found"] and c_val >= CONF_THRESHOLD and s.get("label") in ("added", "removed"):
        s["label"] = "present_both"
    return s

def sam_labels(summaries: dict) -> dict:
    return {c: s.get("label", "not_found") for c, s in summaries.items()}

def _norm(s: str) -> str:
    return " ".join(str(s).lower().strip().split())

def _find(concept: str, summaries: dict):
    n = _norm(concept)
    if n in summaries:
        return n, summaries[n]
    best_key, best_v, best_overlap = None, None, 0
    for k, v in summaries.items():
        overlap = len(set(_norm(k).split()) & set(n.split()))
        if overlap > best_overlap:
            best_key, best_v, best_overlap = k, v, overlap
    if best_key is not None:
        return best_key, best_v
    return None, None

def _evidence(concept: str, s: dict) -> dict:
    return {"concept": concept, "label": s.get("label"),
            "original_label": s.get("original_label", s.get("label")),
            "recheck": s.get("recheck"),
            "frames_before": s["before"]["frames_detected"], "frames_after": s["after"]["frames_detected"],
            "best_box_before": s["before"]["best_box"], "best_box_after": s["after"]["best_box"],
            "best_frame_before": s["before"]["best_frame"], "best_frame_after": s["after"]["best_frame"],
            "confidence_before": s["before"]["best_conf"], "confidence_after": s["after"]["best_conf"],
            "mean_conf_before": s["before"]["mean_conf"], "mean_conf_after": s["after"]["mean_conf"],
            "max_instances_before": s["before"]["max_instances"], "max_instances_after": s["after"]["max_instances"]}

def _recheck_reason(key: str, s: dict, change: str) -> str | None:
    """Specific human-readable reason when a re-check overturns a claim."""
    r = s.get("recheck") or {}
    if not r.get("found"):
        return None
    try:
        c_val = float(r.get("c", 0.0) or 0.0)
    except Exception:
        c_val = 0.0
    video = str(r.get("video", "")).lower()
    side = "BEFORE" if video == "before" else "AFTER" if video == "after" else video.upper()
    idx = r.get("keyframe_index")
    if change == "added" and video == "before":
        return (f"SAM image re-check found '{key}' in BEFORE keyframe {idx} "
                f"(c={c_val:.2f}) - not added")
    if change == "removed" and video == "after":
        return (f"SAM image re-check found '{key}' in AFTER keyframe {idx} "
                f"(c={c_val:.2f}) - not removed")
    return None

def _clean_watchlist(watchlist) -> list[str]:
    if not watchlist:
        return []
    if isinstance(watchlist, str):
        watchlist = [c.strip() for c in watchlist.split(",")]
    out, seen = [], set()
    for item in watchlist or []:
        c = _norm(item)
        if c and c not in seen:
            seen.add(c); out.append(c)
    return out

def _is_watchlist(concept: str, watchlist_norm: list[str]) -> bool:
    """Case-insensitive watchlist match with token-overlap fuzziness."""
    n = _norm(concept)
    if not n or not watchlist_norm:
        return False
    if n in watchlist_norm:
        return True
    n_tokens = set(n.split())
    for w in watchlist_norm:
        if n_tokens & set(w.split()):
            return True
    return False

def _clean_risk(value) -> str | None:
    v = str(value or "").strip().lower()
    return v if v in ("high", "medium", "low") else None

def _claim_risk(claim: dict, concept: str, watchlist_norm: list[str]) -> tuple[str, str]:
    """Risk carried from Spark, with a defensive fallback."""
    risk = _clean_risk(claim.get("risk"))
    reason = str(claim.get("risk_reason") or "")
    if risk is not None:
        return risk, reason
    # Spark omitted/invalid risk: watchlist changes are safety-relevant.
    if _is_watchlist(concept, watchlist_norm):
        return "high", reason or f"'{concept}' is on the caregiver watchlist, so any change to it is treated as high risk."
    return "medium", reason

def _sam_only_risk(concept: str, change: str, watchlist_norm: list[str]) -> tuple[str, str]:
    """Simple risk rule for SAM-only kept claims (no Spark judgement)."""
    if _is_watchlist(concept, watchlist_norm):
        return "high", (f"'{concept}' is on the caregiver watchlist and SAM detected it as {change}; "
                        "check it is within reach and not a trip hazard.")
    return "medium", (f"SAM detected '{concept}' as {change} but Spark did not assess it; "
                      "defaulting to medium risk pending a caregiver check.")

def grounding_filter(raw_claims: list, summaries: dict, moved_verdicts: dict | None = None,
                     watchlist=None) -> tuple[list, list]:
    """Keep only claims SAM supports; add SAM-only added/removed claims.

    'moved' requires present_both AND a moved verdict with
    confidence >= MOVED_MIN_CONF.

    ``risk``/``risk_reason`` from Spark (see spark.write_claims) are
    carried through onto BOTH kept and dropped claims. SAM-only kept
    claims get risk from a simple rule: watchlist item -> high,
    otherwise medium.
    """
    moved_verdicts = moved_verdicts or {}
    watchlist_norm = _clean_watchlist(watchlist)
    kept, dropped, kept_concepts = [], [], set()
    for claim in raw_claims or []:
        if not isinstance(claim, dict):
            dropped.append({"concept": str(claim), "change": None, "explanation": "", "safety_note": None,
                            "risk": "medium", "risk_reason": "",
                            "reason": "malformed claim (not an object)"}); continue
        concept_raw, change = str(claim.get("concept", "")), claim.get("change")
        key, s = _find(concept_raw, summaries)
        risk, risk_reason = _claim_risk(claim, key or concept_raw, watchlist_norm)
        if s is None:
            dropped.append({**claim, "concept": concept_raw, "risk": risk, "risk_reason": risk_reason,
                            "reason": f"no SAM evidence: concept '{concept_raw}' was not tracked"}); continue
        label = s["label"]
        ok, reason = False, ""
        if change == "added":
            ok = label == "added"
            reason = _recheck_reason(key, s, "added") or (
                f"SAM label is '{label}', not 'added' for '{key}'")
        elif change == "removed":
            ok = label == "removed"
            reason = _recheck_reason(key, s, "removed") or (
                f"SAM label is '{label}', not 'removed' for '{key}'")
        elif change == "moved":
            v = moved_verdicts.get(key) or moved_verdicts.get(concept_raw) or {}
            try:
                v_conf = float(v.get("confidence", 0.0) or 0.0)
            except Exception:
                v_conf = 0.0
            v_moved = bool(v.get("moved"))
            if label != "present_both":
                rc = _recheck_reason(key, s, "moved")
                reason = rc or f"SAM label is '{label}', not 'present_both' for '{key}' - not moved"
            elif not v_moved:
                detail = f": {v.get('reason')}" if v.get("reason") else ""
                reason = (f"moved-judge said not moved for '{key}' "
                          f"(confidence {v_conf:.2f}){detail}")
            elif v_conf < MOVED_MIN_CONF:
                reason = f"moved-judge confidence {v_conf:.2f} < {MOVED_MIN_CONF:.2f}"
            else:
                ok = True
        else:
            reason = f"unsupported change type '{change}'"
        if ok:
            kept.append({"concept": key, "change": change, "explanation": claim.get("explanation", ""),
                         "safety_note": claim.get("safety_note"), "source": "spark",
                         "risk": risk, "risk_reason": risk_reason,
                         "evidence": _evidence(key, s)})
            kept_concepts.add(key)
        else:
            dropped.append({**claim, "concept": key, "risk": risk, "risk_reason": risk_reason,
                            "reason": reason})
    # SAM-only added/removed that Spark missed
    for concept, s in summaries.items():
        if s["label"] in ("added", "removed") and concept not in kept_concepts:
            risk, risk_reason = _sam_only_risk(concept, s["label"], watchlist_norm)
            kept.append({"concept": concept, "change": s["label"],
                         "explanation": f"SAM 3.1 detected '{concept}' as {s['label']} (present in "
                                        f"{s['after']['frames_detected'] if s['label']=='added' else s['before']['frames_detected']} frames).",
                         "safety_note": None, "source": "sam",
                         "risk": risk, "risk_reason": risk_reason,
                         "evidence": _evidence(concept, s)})
            kept_concepts.add(concept)
    return kept, dropped

def _safe(s: str) -> str:
    return "".join(c if c.isalnum() else "_" for c in str(s).lower()).strip("_")

def _norm_concept(s: str) -> str:
    return " ".join(str(s).strip().lower().split())

def _clean_list(items) -> list[str]:
    """Accept None / comma string / iterable; return stripped strings."""
    if not items:
        return []
    if isinstance(items, str):
        items = [c.strip() for c in items.split(",")]
    out = []
    for item in items:
        c = str(item).strip()
        if c:
            out.append(c)
    return out

def _dedupe_prepend(priority: list[str], rest: list[str], cap: int = MAX_CONCEPTS) -> list[str]:
    """priority first, then rest; dedupe case-insensitively; cap total."""
    out, seen = [], set()
    for c in list(priority) + list(rest):
        key = _norm_concept(c)
        if not key or key in seen:
            continue
        seen.add(key); out.append(str(c).strip())
        if len(out) >= cap:
            break
    return out

def _find_summary(concept: str, summaries: dict):
    """Exact (normalised) match, else best token-overlap match."""
    n = _norm_concept(concept)
    for k, v in summaries.items():
        if _norm_concept(k) == n:
            return k, v
    best_key, best_v, best_overlap = None, None, 0
    for k, v in summaries.items():
        overlap = len(set(_norm_concept(k).split()) & set(n.split()))
        if overlap > best_overlap:
            best_key, best_v, best_overlap = k, v, overlap
    return best_key, best_v

def build_report(result: dict) -> str:
    lines = ["# SplatDiff — grounded change report", "", result.get("summary") or "", "", "## Kept claims (SAM-grounded)"]
    if not result.get("claims"):
        lines.append("- No grounded changes found.")
    for c in result.get("claims", []):
        ev = c.get("evidence", {})
        try:
            conf_b = float(ev.get("confidence_before", 0) or 0)
        except Exception:
            conf_b = 0.0
        try:
            conf_a = float(ev.get("confidence_after", 0) or 0)
        except Exception:
            conf_a = 0.0
        lines.append(f"- **{c['concept']} — {c['change']}** ({c.get('source','')}) "
                     f"[risk: {c.get('risk','')}] : {c.get('explanation','')} "
                     f"[frames before={ev.get('frames_before')} after={ev.get('frames_after')}, "
                     f"conf before={conf_b:.2f} after={conf_a:.2f}]")
        if c.get("risk_reason"):
            lines.append(f"  - Risk: {c.get('risk','')} — {c['risk_reason']}")
        if c.get("safety_note"):
            lines.append(f"  - Safety note: {c['safety_note']}")
    if result.get("watchlist_status"):
        lines += ["", "## Watchlist"]
        for w in result["watchlist_status"]:
            lines.append(f"- **{w.get('concept','')} — {w.get('status','')}** "
                         f"[risk: {w.get('risk','')}] : {w.get('note','')}")
    lines += ["", "## Dropped claims (not supported by SAM)"]
    if not result.get("dropped"):
        lines.append("- None.")
    for d in result.get("dropped", []):
        lines.append(f"- ~~{d.get('concept','')} — {d.get('change','')}~~ "
                     f"[risk: {d.get('risk','')}] — dropped: {d.get('reason','')}")
    lines += ["", f"Cost estimate (ledger total): ${result.get('cost_usd',0):.4f} · Latency: {result.get('latency_s',0):.1f}s"]
    return "\n".join(lines) + "\n"

def _best_detection(parsed) -> dict | None:
    """Highest-confidence detection (c >= threshold) in segment_image output."""
    best = None
    if isinstance(parsed, (list, tuple)):
        iterable = [parsed]
    elif isinstance(parsed, dict):
        iterable = list(parsed.values())
    else:
        iterable = []
    for dets in iterable:
        if isinstance(dets, dict):
            dets = [dets]
        for d in dets or []:
            try:
                c = float(d.get("c", 0))
            except Exception:
                continue
            if c >= diffmod.CONF_THRESHOLD and (best is None or c > best["c"]):
                box = d.get("box")
                best = {"box": list(box) if box is not None else None, "c": c}
    return best

def _run_image_rechecks(summaries: dict, before_keyframes: list, after_keyframes: list, prog,
                        emit=None) -> int:
    """SAM image re-check for added/removed concepts on the 'absent' side.

    Mutates summaries: stores per-concept 'recheck' info and flips the
    label to 'present_both' when a keyframe detection overturns the
    video-tracking label. Original label is kept as 'original_label'.

    Emits recheck start/progress/done events (if ``emit`` is given) as
    each concept finishes. Returns the number of SAM image requests made.
    """
    def _emit(ev: dict):
        if emit is not None:
            try:
                emit(ev)
            except Exception:
                log.exception("on_event callback failed for %s", ev)

    candidates = [(c, s) for c, s in summaries.items() if s.get("label") in ("added", "removed")]
    _emit({"stage": "recheck", "status": "start", "concept": "", "found": False, "c": 0.0})
    if not candidates:
        _emit({"stage": "recheck", "status": "done", "concept": "", "found": False, "c": 0.0})
        return 0
    prog(f"SAM image re-check for {len(candidates)} added/removed concept(s)…")
    absent_side = {}
    for concept, s in candidates:
        # added -> supposedly absent in BEFORE; removed -> absent in AFTER
        absent_side[concept] = "before" if s["label"] == "added" else "after"

    def _job(item):
        concept, s = item
        side = absent_side[concept]
        keyframes = before_keyframes if side == "before" else after_keyframes
        best, calls = None, 0
        for idx, kf_path in enumerate(keyframes or []):
            calls += 1
            try:
                parsed = segment_image(kf_path, concept)
            except Exception as e:
                log.warning("SAM image re-check failed for %s keyframe %s: %s", concept, idx, e)
                continue
            det = _best_detection(parsed)
            if det is not None and (best is None or det["c"] > best["c"]):
                best = {"video": side, "keyframe_index": idx, "box": det["box"], "c": det["c"]}
        return concept, side, best, calls

    found: dict[str, dict] = {}
    total_calls = 0
    with ThreadPoolExecutor(max_workers=6) as ex:
        futures = {ex.submit(_job, item): item[0] for item in candidates}
        for fut in as_completed(futures):
            concept = futures[fut]
            try:
                concept, side, det, calls = fut.result()
            except Exception as e:
                log.exception("SAM image re-check job failed for %s", concept)
                det, calls = None, 0
            total_calls += calls
            if det is not None:
                found[concept] = det
            _emit({"stage": "recheck", "status": "progress", "concept": concept,
                   "found": det is not None,
                   "c": float(det["c"]) if det is not None else 0.0})
    for concept, s in candidates:
        side = absent_side[concept]
        keyframes = before_keyframes if side == "before" else after_keyframes
        if concept in found:
            det = found[concept]
            diffmod.apply_image_recheck(summaries, concept,
                                        {"checked": True, "found": True, "video": side,
                                         "keyframe_index": det["keyframe_index"],
                                         "box": det["box"], "c": det["c"]})
            prog(f"Re-check: SAM image found '{concept}' in {side.upper()} "
                 f"keyframe {det['keyframe_index']} (c={det['c']:.2f}) - label now 'present_both' (was '{s.get('original_label')}')")
        else:
            diffmod.apply_image_recheck(summaries, concept,
                                        {"checked": True, "found": False, "video": side,
                                         "keyframe_index": None, "box": None, "c": 0.0,
                                         "keyframes_checked": len(keyframes or [])})
    _emit({"stage": "recheck", "status": "done", "concept": "", "found": False, "c": 0.0})
    return total_calls

def _judge_frame(side: str, s: dict, norm_path, keyframes: list):
    """Return (frame_ndarray, box) for the moved judge for one side.

    If the image re-check found this concept on this (supposedly absent)
    side, use that keyframe and its box; otherwise use the best tracked
    video frame.
    """
    recheck = s.get("recheck") or {}
    if recheck.get("found") and str(recheck.get("video", "")).lower() == side:
        idx = recheck.get("keyframe_index")
        if idx is not None and keyframes and 0 <= int(idx) < len(keyframes):
            frame = cv2.imread(str(keyframes[int(idx)]))
            if frame is not None:
                return frame, recheck.get("box")
        log.warning("re-check keyframe unavailable for %s side %s; falling back to video frame",
                    s.get("concept"), side)
    summ = s[side]
    frame = read_frame(norm_path, summ["best_frame"] or 0) if summ["best_frame"] is not None else read_frame(norm_path, 0)
    return frame, summ["best_box"]

def _watchlist_note(concept: str, status: str) -> str:
    c = str(concept)
    if status == "missing":
        return (f"{c} is no longer where it was - check it is within reach of the bed/chair "
                f"and find it before it is needed.")
    if status == "moved":
        return (f"{c} has moved from where it was - check it is within reach of the bed/chair "
                f"and has not become a trip hazard.")
    if status == "not_seen":
        return (f"{c} was not detected in either video - check it manually; SAM may have missed it, "
                f"or it may have been removed before the BEFORE video.")
    if status == "appeared":
        return (f"{c} was not seen in the BEFORE video but appears in the AFTER video - check it is "
                f"stored safely, within reach if it is needed, and not blocking a walkway.")
    return f"{c} appears to still be in place."

_WATCHLIST_RISK = {"missing": "high", "moved": "high", "not_seen": "medium",
                   "appeared": "medium", "in_place": "low"}

def _build_watchlist_status(watchlist_list: list[str], summaries: dict, kept: list,
                            moved_verdicts: dict, images_for) -> list[dict]:
    """Derive per-watchlist-item status from SAM labels + moved verdicts."""
    moved_kept = {_norm_concept(c.get("concept", "")) for c in kept if c.get("change") == "moved"}
    out = []
    for item in watchlist_list:
        key, s = _find_summary(item, summaries)
        concept = key or _norm_concept(item)
        if s is None:
            status = "not_seen"
        else:
            label = s.get("label", "not_found")
            if label == "removed":
                status = "missing"
            elif label == "added":
                status = "appeared"
            elif label == "present_both":
                # 'moved verdict kept' = a kept moved claim for this concept;
                # equivalently a true verdict at/above the grounding threshold.
                v = moved_verdicts.get(key) or {}
                try:
                    v_conf = float(v.get("confidence", 0.0) or 0.0)
                except Exception:
                    v_conf = 0.0
                verdict_kept = bool(v.get("moved")) and v_conf >= diffmod.MOVED_MIN_CONF
                status = "moved" if (_norm_concept(key) in moved_kept or verdict_kept) else "in_place"
            else:  # not_found
                status = "not_seen"
        out.append({"concept": concept, "status": status,
                    "risk": _WATCHLIST_RISK.get(status, "medium"),
                    "note": _watchlist_note(concept, status),
                    "images": images_for(concept)})
    return out

def run(before, after, concepts=None, out_dir="outputs", progress=None, on_event=None, watchlist=None) -> dict:
    def prog(msg: str):
        log.info(msg)
        if progress:
            progress(msg)

    start, cost_start = time.time(), total_cost()
    run_id = datetime.now().strftime("%Y%m%d_%H%M%S")
    watchlist_list = _clean_list(watchlist)
    event_lock = threading.Lock()

    def _running_cost() -> float:
        try:
            return round(float(total_cost()) - float(cost_start), 6)
        except Exception:
            return 0.0

    def emit(ev: dict):
        """Thread-safe structured event: adds elapsed_s + running cost_usd."""
        if on_event is None:
            return
        event = dict(ev)
        event["elapsed_s"] = round(time.time() - start, 3)
        event["cost_usd"] = _running_cost()
        with event_lock:
            try:
                on_event(event)
            except Exception:
                log.exception("on_event callback failed for %s", event)

    try:
        out = Path(out_dir) / run_id; out.mkdir(parents=True, exist_ok=True)

        def _url(filename: str) -> str:
            return f"/outputs/{run_id}/{filename}"

        def _images_for(concept: str) -> dict:
            safe = _safe(concept)
            b_name, a_name = f"before_{safe}.jpg", f"after_{safe}.jpg"
            return {"before": _url(b_name) if (out / b_name).exists() else None,
                    "after": _url(a_name) if (out / a_name).exists() else None}

        # 1. Normalize
        emit({"stage": "normalize", "status": "start"})
        prog("Normalising videos (ffmpeg, 5 fps, max 30 s)…")
        before_norm, after_norm = normalize_video(before), normalize_video(after)
        emit({"stage": "normalize", "status": "done"})

        # 2. Keyframes
        prog("Extracting keyframes…")
        bf = extract_keyframes(before_norm, 6, out / "keyframes_before")
        af = extract_keyframes(after_norm, 6, out / "keyframes_after")
        keyframe_urls = {
            "before": [f"/outputs/{run_id}/keyframes_before/{Path(p).name}" for p in (bf or [])],
            "after": [f"/outputs/{run_id}/keyframes_after/{Path(p).name}" for p in (af or [])],
        }
        emit({"stage": "keyframes", "status": "done",
              "before": list(keyframe_urls["before"]), "after": list(keyframe_urls["after"])})

        # 3. Inventory (watchlist always tracked: prepended, capped)
        emit({"stage": "inventory", "status": "start", "model": "muse-spark-1.3",
              "concepts": [], "objects": []})
        prog("Inventorying objects with Muse Spark 1.3…")
        inventory_called = not bool(concepts)
        objects = list_objects(bf, af, concepts=concepts)
        inventory_concepts = [o["concept"] for o in objects if isinstance(o, dict) and o.get("concept")]
        concept_list = _dedupe_prepend(watchlist_list, inventory_concepts, cap=MAX_CONCEPTS)
        # Ensure watchlist items are represented in `objects` too.
        known = {_norm_concept(o.get("concept", "")) for o in objects if isinstance(o, dict)}
        extra_objects = [{"concept": c, "where_before": "", "where_after": "",
                          "suspected_change": "unchanged"}
                         for c in concept_list if _norm_concept(c) not in known]
        if extra_objects:
            objects = extra_objects + list(objects)
        emit({"stage": "inventory", "status": "done", "model": "muse-spark-1.3",
              "concepts": list(concept_list), "objects": objects})
        prog(f"Concepts ({len(concept_list)}): {', '.join(concept_list) or 'none'}")

        # 4. SAM tracking (concurrent; progress as each future completes)
        prog("Tracking concepts with SAM 3.1 (concurrent)…")
        tasks = [(c, "before", before_norm) for c in concept_list] + [(c, "after", after_norm) for c in concept_list]
        emit({"stage": "track", "status": "start", "total": len(tasks)})
        before_tracks, after_tracks = {}, {}
        sam_requests, sam_frames = 0, 0

        def _job(t):
            concept, side, path = t
            try:
                return concept, side, track_concept(path, concept)
            except Exception as e:
                log.exception("SAM failed for %s/%s", concept, side); return concept, side, {}

        done_count = 0
        if tasks:
            with ThreadPoolExecutor(max_workers=6) as ex:
                futures = {ex.submit(_job, t): t for t in tasks}
                for fut in as_completed(futures):
                    concept, side, parsed = fut.result()
                    parsed = parsed or {}
                    (before_tracks if side == "before" else after_tracks)[concept] = {
                        int(k): v for k, v in parsed.items()}
                    done_count += 1; sam_requests += 1
                    try:
                        n_frames = len(parsed)
                        n_dets = sum(len(v or []) for v in parsed.values())
                    except Exception:
                        n_frames, n_dets = 0, 0
                    sam_frames += n_frames
                    emit({"stage": "track", "status": "progress", "concept": concept,
                          "video": side, "frames": int(n_frames), "detections": int(n_dets),
                          "done": done_count, "total": len(tasks)})
        emit({"stage": "track", "status": "done"})
        summaries = diffmod.summarize_tracks(before_tracks, after_tracks)
        # Snapshot of ORIGINAL video-tracking labels: the final write_claims
        # call below must still see these (not re-check-corrected labels), so
        # Spark may e.g. claim 'armchair added' and the grounding filter can
        # visibly drop it citing the re-check.
        original_summaries = copy.deepcopy(summaries)

        # 5. SAM image re-checks (emits its own start/progress/done events)
        recheck_calls = _run_image_rechecks(summaries, bf, af, prog, emit=emit)
        rechecks_count = sum(1 for s in summaries.values() if (s.get("recheck") or {}).get("checked"))

        # 6. Moved judge (concurrent; progress as each future completes)
        prog("Judging moved objects with Muse Spark 1.3…")
        emit({"stage": "judge", "status": "start", "concept": "", "moved": False,
              "confidence": 0.0, "reason": ""})
        moved_verdicts = {}
        judge_items = [(concept, s) for concept, s in summaries.items() if s["label"] == "present_both"]
        if judge_items:
            prog(f"Judging moved objects concurrently for {len(judge_items)} concept(s)…")

        def _judge_job(item):
            concept, s = item
            bp = ap = None
            try:
                bframe, bbox = _judge_frame("before", s, before_norm, bf)
                aframe, abox = _judge_frame("after", s, after_norm, af)
                if bframe is None or aframe is None:
                    return concept, {"moved": False, "confidence": 0.0, "reason": "best frame unavailable"}
                bp, ap = out / f"_judge_{_safe(concept)}_before.jpg", out / f"_judge_{_safe(concept)}_after.jpg"
                cv2.imwrite(str(bp), bframe); cv2.imwrite(str(ap), aframe)
                try:
                    verdict = judge_moved(concept, str(bp), str(ap), bbox, abox)
                except Exception as e:
                    log.exception("judge_moved failed %s", concept)
                    verdict = {"moved": False, "confidence": 0.0, "reason": str(e)}
                return concept, verdict
            except Exception as e:
                log.exception("judge_moved failed %s", concept)
                return concept, {"moved": False, "confidence": 0.0, "reason": str(e)}
            finally:
                # judge temp files are per-concept, so concurrent cleanup is safe.
                for _p in (bp, ap):
                    if _p is not None:
                        try:
                            _p.unlink(missing_ok=True)
                        except Exception:
                            pass

        if judge_items:
            with ThreadPoolExecutor(max_workers=6) as ex:
                futures = {ex.submit(_judge_job, item): item[0] for item in judge_items}
                for fut in as_completed(futures):
                    concept = futures[fut]
                    try:
                        concept, verdict = fut.result()
                    except Exception as e:
                        log.exception("judge job failed %s", concept)
                        verdict = {"moved": False, "confidence": 0.0, "reason": str(e)}
                    moved_verdicts[concept] = verdict
                    try:
                        v_conf = float(verdict.get("confidence", 0.0) or 0.0)
                    except Exception:
                        v_conf = 0.0
                    prog(f"Judged '{concept}': moved={bool(verdict.get('moved'))} (confidence {v_conf:.2f})")
                    emit({"stage": "judge", "status": "progress", "concept": concept,
                          "moved": bool(verdict.get("moved")), "confidence": v_conf,
                          "reason": str(verdict.get("reason", ""))})
        emit({"stage": "judge", "status": "done", "concept": "", "moved": False,
              "confidence": 0.0, "reason": ""})

        # 7. Final Spark report
        emit({"stage": "report", "status": "start", "model": "muse-spark-1.3", "raw_claims": 0})
        prog("Writing grounded report with Muse Spark 1.3…")
        # Deliberately pass original (pre-recheck) summaries: raw Spark claims
        # stay separate from, and are then checked against, corrected evidence.
        written = write_claims(original_summaries, moved_verdicts, watchlist=watchlist_list)
        raw_claims = written.get("claims", [])
        emit({"stage": "report", "status": "done", "model": "muse-spark-1.3",
              "raw_claims": len(raw_claims)})

        # 8. Grounding filter (carries risk/risk_reason through)
        kept, dropped = diffmod.grounding_filter(raw_claims, summaries, moved_verdicts,
                                                 watchlist=watchlist_list)
        emit({"stage": "filter", "status": "done", "kept": len(kept), "dropped": len(dropped)})

        # Effective change per concept for annotation colours
        eff = {c["concept"]: c["change"] for c in kept}
        annotated = []
        prog("Saving annotated keyframes and outputs…")
        for concept, s in summaries.items():
            change = eff.get(concept)
            color = COLORS.get(change, (180, 180, 180))
            label_text = f"{concept}: {change}" if change else f"{concept}: {s['label']}"
            for side, norm, summ, keyframes in (("before", before_norm, s["before"], bf), ("after", after_norm, s["after"], af)):
                recheck = s.get("recheck") or {}
                is_recheck_side = bool(recheck.get("found") and str(recheck.get("video", "")).lower() == side
                          and recheck.get("keyframe_index") is not None and recheck.get("box") is not None)
                if summ["best_frame"] is None or summ["best_box"] is None:
                    if not is_recheck_side:
                        continue
                    # Annotate the re-check keyframe that overturned the label.
                    if change == "added" and side == "before":
                        continue
                    if change == "removed" and side == "after":
                        continue
                    idx = int(recheck["keyframe_index"])
                    if not (0 <= idx < len(keyframes or [])):
                        continue
                    frame = cv2.imread(str(keyframes[idx]))
                    if frame is None:
                        continue
                    p = out / f"{side}_{_safe(concept)}.jpg"
                    cv2.imwrite(str(p), draw_box(frame, recheck["box"], label_text, color))
                    annotated.append(str(p))
                    continue
                if change == "added" and side == "before":
                    continue
                if change == "removed" and side == "after":
                    continue
                frame = read_frame(norm, summ["best_frame"])
                if frame is None:
                    continue
                p = out / f"{side}_{_safe(concept)}.jpg"
                cv2.imwrite(str(p), draw_box(frame, summ["best_box"], label_text, color))
                annotated.append(str(p))

        # Evidence images (annotated frames) on every kept + dropped claim.
        for claim in list(kept) + list(dropped):
            claim["images"] = _images_for(str(claim.get("concept", "")))
            claim.setdefault("risk", "medium")
            claim.setdefault("risk_reason", "")

        watchlist_status = _build_watchlist_status(watchlist_list, summaries, kept,
                                                   moved_verdicts, _images_for)

        spark_calls = (1 if inventory_called else 0) + len(judge_items) + 1
        stats = {"concepts": len(concept_list), "sam_requests": int(sam_requests),
                 "sam_frames": int(sam_frames), "spark_calls": int(spark_calls),
                 "rechecks": int(rechecks_count)}
        models = ["muse-spark-1.3", "sam-3.1"]

        result = {"run_id": run_id, "before": str(before), "after": str(after),
                  "normalized_before": before_norm, "normalized_after": after_norm,
                  "objects": objects, "concepts": concept_list, "summaries": summaries,
                  "diff_table": summaries, "moved_verdicts": moved_verdicts,
                  "summary": written.get("summary", ""), "raw_claims": raw_claims,
                  "claims": kept, "kept_claims": kept, "dropped": dropped,
                  "annotated_images": annotated,
                  "watchlist": list(watchlist_list), "watchlist_status": watchlist_status,
                  "keyframes": keyframe_urls, "stats": stats, "models": models,
                  "latency_s": round(time.time() - start, 3),
                  "cost_usd": round(total_cost() - cost_start, 6)}
        (out / "result.json").write_text(json.dumps(result, indent=2, default=str), encoding="utf-8")
        (out / "report.md").write_text(build_report(result), encoding="utf-8")
        result["result_dir"] = str(out); result["report"] = build_report(result)
        prog(f"Done in {result['latency_s']:.1f}s — {len(kept)} kept, {len(dropped)} dropped.")
        emit({"stage": "done", "status": "done", "run_id": run_id})
        return result
    except Exception as e:
        log.exception("pipeline run %s failed", run_id)
        emit({"stage": "error", "status": "error", "message": str(e)})
        raise

def main(argv=None):
    ap = argparse.ArgumentParser(prog="splatdiff")
    ap.add_argument("before"); ap.add_argument("after")
    ap.add_argument("--concepts", default=None, help="comma list override")
    ap.add_argument("--watchlist", default=None, help="comma list of care items that must stay put")
    ap.add_argument("--out", default="outputs")
    args = ap.parse_args(argv)
    concepts = [c.strip() for c in args.concepts.split(",")] if args.concepts else None
    watchlist = [c.strip() for c in args.watchlist.split(",") if c.strip()] if args.watchlist else None
    result = run(args.before, args.after, concepts=concepts, out_dir=args.out, watchlist=watchlist)
    print(f"\n{'Concept':<24} {'Change':<10} {'Risk':<8} Source")
    print("-" * 60)
    for c in result["claims"]:
        print(f"{c['concept']:<24} {c['change']:<10} {c.get('risk',''):<8} {c.get('source','')}")
    for d in result["dropped"]:
        print(f"{d.get('concept',''):<24} {d.get('change',''):<10} {'':<8} DROPPED: {d.get('reason','')}")
    if result.get("watchlist_status"):
        print("\nWatchlist:")
        for w in result["watchlist_status"]:
            print(f"  {w['concept']:<24} {w['status']:<10} risk={w['risk']}")
    print(f"\nSummary: {result.get('summary','')}")
    print(f"Saved to {result.get('result_dir','outputs/'+result['run_id'])} | cost ${result['cost_usd']:.4f} | {result['latency_s']:.1f}s")

if __name__ == "__main__":
    main()
