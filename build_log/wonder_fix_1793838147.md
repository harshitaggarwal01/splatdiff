def _room_slug(room) -> str:
    """Validate / derive a room slug: lowercase, [a-z0-9-], max 40 chars."""
    raw = str(room or "").strip().lower()
    slug = re.sub(r"[^a-z0-9-]+", "-", raw)
    slug = re.sub(r"-{2,}", "-", slug).strip("-")[:40].strip("-")
    if not slug or not _SLUG_RE.match(slug):
        return ""
    return slug


def _now_iso() -> str:
    return datetime.now().astimezone().isoformat()


def _parse_date(value, fallback_path: Optional[Path] = None):
    """Best-effort parse of a date/run_id into an aware datetime."""
    if isinstance(value, str) and value.strip():
        txt = value.strip()
        for fmt in ("%Y%m%d_%H%M%S", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S", "%Y-%m-%d"):
            try:
                return datetime.strptime(txt[:19] if fmt != "%Y%m%d_%H%M%S" else txt[:15], fmt).astimezone()
            except Exception:
                pass
        try:
            dt = datetime.fromisoformat(txt.replace("Z", "+00:00"))
            return dt if dt.tzinfo else dt.astimezone()
        except Exception:
            pass
        # run_id embedded at start
        m = re.match(r"^(\d{8}_\d{6})", txt)
        if m:
            try:
                return datetime.strptime(m.group(1), "%Y%m%d_%H%M%S").astimezone()
            except Exception:
                pass
    if fallback_path is not None:
        try:
            return datetime.fromtimestamp(fallback_path.stat().st_mtime).astimezone()
        except Exception:
            pass
    return None


def _result_datetime(result: dict, path: Optional[Path] = None):
    if isinstance(result, dict):
        for key in ("ts", "date", "created_at", "timestamp", "run_id"):
            dt = _parse_date(result.get(key))
            if dt is not None:
                return dt
    return _parse_date(None, fallback_path=path) or datetime.now().astimezone()


def _watchlist_entries(result: dict):
    """Return list of watchlist item dicts from any plausible result shape."""
    if not isinstance(result, dict):
        return []
    candidates = []
    for key in ("watchlist_results", "watchlist_status", "watched_items", "watchlist_items"):
        val = result.get(key)
        if isinstance(val, list):
            candidates.extend([v for v in val if isinstance(v, dict)])
        elif isinstance(val, dict):
            candidates.extend([v for v in val.values() if isinstance(v, dict)])
    val = result.get("watchlist")
    if isinstance(val, list):
        for v in val:
            if isinstance(v, dict):
                candidates.append(v)
    elif isinstance(val, dict):
        for k in ("items", "results", "entries"):
            sub = val.get(k)
            if isinstance(sub, list):
                candidates.extend([v for v in sub if isinstance(v, dict)])
    return candidates


def _watchlist_names(result: dict):
    names = []
    if not isinstance(result, dict):
        return names
    for entry in _watchlist_entries(result):
        name = entry.get("concept") or entry.get("name") or entry.get("item")
        if name:
            names.append(str(name))
    val = result.get("watchlist")
    if isinstance(val, list):
        for v in val:
            if isinstance(v, str) and v.strip():
                names.append(v.strip())
    elif isinstance(val, dict):
        for k in ("items", "concepts", "names"):
            sub = val.get(k)
            if isinstance(sub, list):
                names.extend([str(v) for v in sub if isinstance(v, str) and v.strip()])
    # de-dupe, preserve order
    seen, out = set(), []
    for n in names:
        key = n.lower()
        if key not in seen:
            seen.add(key)
            out.append(n)
    return out


def _kept_claims(result: dict):
    if not isinstance(result, dict):
        return []
    claims = result.get("kept_claims") or result.get("claims") or []
    return [c for c in claims if isinstance(c, dict)] if isinstance(claims, list) else []


def _next_step(change, risk=None, intentional=False) -> str:
    """Change-specific plain next step for a family-feed item."""
    if intentional:
        return "Dad says this was on purpose — consider making it the new setup."
    c = str(change or "").strip().lower()
    if c == "moved":
        return "Ask Dad if he moved it on purpose; if not, someone there puts it back with him."
    if c in ("removed", "missing"):
        return "Ask Dad where it is; put it back where he expects it."
    if c in ("added", "appeared", "on_floor"):
        return "Move it out of the walking path, with Dad."
    if c == "not_seen":
        return "Check it in person next visit."
    return "Have a look at this on the next visit."


def _result_trend_text(result) -> Optional[str]:
    """Trend sentence (or None) from a pipeline result's 'trend' value."""
    if not isinstance(result, dict):
        return None
    trend = result.get("trend")
    if isinstance(trend, dict):
        text = trend.get("trend")
        return str(text) if text else None
    if isinstance(trend, str) and trend.strip():
        return trend
    return None


def _feed_status(result: dict) -> str:
    coverage = result.get("coverage") if isinstance(result, dict) else None
    if isinstance(coverage, dict) and coverage.get("ok") is False:
        return "coverage_gap"
    for claim in _kept_claims(result):
        if str(claim.get("risk") or "").strip().lower() == "high":
            return "needs_attention"
    for entry in _watchlist_entries(result):
        status = str(entry.get("status") or entry.get("state") or "").strip().lower()
        if status in ("moved", "missing", "not_seen"):
            return "needs_attention"
    return "all_clear"


def _feed_entry_from_result(result: dict, path: Optional[Path] = None) -> dict:
    room = str(result.get("room") or "Bedroom") if isinstance(result, dict) else "Bedroom"
    # Tolerate run_id=None (e.g. glasses entries / synthetic results): fall
    # back to the outputs folder name or a random id, never the string "None".
    raw_run_id = result.get("run_id") if isinstance(result, dict) else None
    run_id = str(raw_run_id) if raw_run_id else (path.parent.name if path else uuid.uuid4().hex[:12])
    dt = _result_datetime(result, path)
    status = _feed_status(result)
    items = []
    # Kept claims first: items are deduped by concept, and the kept claim
    # for a concept is preferred over its watchlist row.
    seen_concepts: set = set()
    for claim in _kept_claims(result):
        concept = str(claim.get("concept") or claim.get("name") or "item")
        key = concept.strip().lower()
        if key in seen_concepts:
            continue
        seen_concepts.add(key)
        change = str(claim.get("change") or claim.get("label") or "")
        risk = claim.get("risk")
        is_intentional = bool(claim.get("intentional"))
        items.append({
            "concept": concept,
            "change": change,
            "risk": risk,
            "next_step": _next_step(change, risk, intentional=is_intentional),
        })
    # Surface concerning watchlist items not already represented.
    for entry in _watchlist_entries(result):
        st = str(entry.get("status") or entry.get("state") or "").strip().lower()
        if st in ("moved", "missing", "not_seen"):
            concept = str(entry.get("concept") or entry.get("name") or entry.get("item") or "watched item")
            if concept.strip().lower() not in seen_concepts:
                items.append({"concept": concept, "change": st, "risk": entry.get("risk"),
                              "next_step": _next_step(st, entry.get("risk"))})
                seen_concepts.add(concept.strip().lower())
    if status == "coverage_gap":
        headline = f"{room} — coverage gap: the walkthroughs didn't cover the same corners"
    elif status == "needs_attention":
        first = items[0] if items else None
        if first:
            headline = f"{room} — needs attention: {first['concept']} {first['change']}".strip()
        else:
            headline = f"{room} — needs attention"
    else:
        if _watchlist_names(result) or _watchlist_entries(result):
            headline = f"{room} — all watched items seen where they were"
        else:
            headline = f"{room} — all clear"
    return {
        "id": run_id,
        "ts": dt.isoformat(),
        "room": room,
        "filmed_by": result.get("filmed_by") if isinstance(result, dict) else None,
        "status": status,
        "headline": headline,
        "items": items,
        "run_id": run_id,
        "trend": _result_trend_text(result),
    }


def _feed_rebuild_from_outputs():
    entries = []
    outputs_dir = ROOT / "outputs"
    if outputs_dir.is_dir():
        files = sorted(outputs_dir.glob("*/result.json"), key=lambda p: p.stat().st_mtime if p.exists() else 0)
        for path in files:
            try:
                result = json.loads(path.read_text(encoding="utf-8"))
                if isinstance(result, dict):
                    entries.append(_feed_entry_from_result(result, path))
            except Exception:
                continue
    return entries


def _feed_persist_all_locked():
    """Rewrite cache/feed.jsonl from FEED. Caller must hold FEED_LOCK."""
    try:
        FEED_PATH.parent.mkdir(parents=True, exist_ok=True)
        tmp = FEED_PATH.with_suffix(".jsonl.tmp")
        with tmp.open("w", encoding="utf-8") as fh:
            for e in FEED:
                fh.write(json.dumps(e, default=str) + "\n")
        tmp.replace(FEED_PATH)
    except Exception as exc:  # noqa: BLE001
        log.warning("could not persist feed: %s", exc)


def _feed_load():
    with FEED_LOCK:
        FEED.clear()
        if FEED_PATH.is_file():
            try:
                for line in FEED_PATH.read_text(encoding="utf-8").splitlines():
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        obj = json.loads(line)
                        if isinstance(obj, dict):
                            # Entries with run_id=None (glasses) load fine.
                            FEED.append(obj)
                    except Exception:
                        continue
                log.info("feed loaded %d entries from %s", len(FEED), FEED_PATH)
                return
            except Exception as exc:  # noqa: BLE001
                log.warning("could not load feed, rebuilding: %s", exc)
                FEED.clear()
        # Missing (or unreadable): rebuild from outputs/*/result.json.
        FEED.extend(_feed_rebuild_from_outputs())
        _feed_persist_all_locked()
        log.info("feed rebuilt %d entries from outputs/", len(FEED))


def _feed_append(entry: dict):
    with FEED_LOCK:
        FEED.append(entry)
        try:
            FEED_PATH.parent.mkdir(parents=True, exist_ok=True)
            with FEED_PATH.open("a", encoding="utf-8") as fh:
                fh.write(json.dumps(entry, default=str) + "\n")
        except Exception as exc:  # noqa: BLE001
            log.warning("could not append to feed file: %s", exc)
    log.info("feed entry appended run_id=%s status=%s", entry.get("run_id"), entry.get("status"))


def _feed_remove(run_id: str):
    # run_id=None entries (glasses) must never be mass-deleted by a
    # history delete; only delete when a concrete run_id is given.
    if not run_id:
        return
    with FEED_LOCK:
        before = len(FEED)
        FEED[:] = [e for e in FEED if e.get("run_id") != run_id and e.get("id") != run_id]
        if len(FEED) != before:
            _feed_persist_all_locked()


def _baseline_meta(slug: str):
    meta_path = BASELINES_DIR / f"{slug}.json"
    video_path = BASELINES_DIR / f"{slug}.mp4"
    if meta_path.is_file():
        try:
            meta = json.loads(meta_path.read_text(encoding="utf-8"))
            if isinstance(meta, dict):
                return meta
        except Exception:
            pass
    if video_path.is_file():
        try:
            dt = datetime.fromtimestamp(video_path.stat().st_mtime).astimezone().isoformat()
        except Exception:
            dt = _now_iso()
        return {"room": slug, "date": dt, "from_run": None, "watchlist": []}
    return None


def _compute_rooms():
    """List of {room, slug, baseline, last_check, days_since, due}."""
    rooms: dict = {}

    def ensure(slug, room_name):
        if slug not in rooms:
            rooms[slug] = {"room": room_name or slug, "slug": slug, "baseline": None, "last_check": None, "days_since": None, "due": True}
        elif room_name:
            rooms[slug]["room"] = room_name
        return rooms[slug]

    if BASELINES_DIR.is_dir():
        for meta_path in sorted(BASELINES_DIR.glob("*.json")):
            slug = meta_path.stem
            if not _SLUG_RE.match(slug):
                continue
            meta = _baseline_meta(slug)
            entry = ensure(slug, (meta or {}).get("room") if isinstance(meta, dict) else slug)
            entry["baseline"] = meta
        for video_path in sorted(BASELINES_DIR.glob("*.mp4")):
            slug = video_path.stem
            if not _SLUG_RE.match(slug):
                continue
            entry = ensure(slug, slug)
            if entry["baseline"] is None:
                entry["baseline"] = _baseline_meta(slug)

    outputs_dir = ROOT / "outputs"
    latest: dict = {}
    if outputs_dir.is_dir():
        files = sorted(outputs_dir.glob("*/result.json"), key=lambda p: p.stat().st_mtime if p.exists() else 0)
        for path in files:
            try:
                result = json.loads(path.read_text(encoding="utf-8"))
            except Exception:
                continue
            if not isinstance(result, dict):
                continue
            room_name = str(result.get("room") or "Bedroom")
            slug = _room_slug(room_name) or "bedroom"
            dt = _result_datetime(result, path)
            if slug not in latest or dt >= latest[slug][0]:
                latest[slug] = (dt, room_name, result, path)
    now = datetime.now().astimezone()
    for slug, (dt, room_name, result, path) in latest.items():
        entry = ensure(slug, room_name)
        run_id = str(result.get("run_id") or path.parent.name)
        entry["last_check"] = {"date": dt.isoformat(), "status": _feed_status(result), "run_id": run_id}
        try:
            days = (now - dt).days
        except Exception:
            days = None
        entry["days_since"] = days
    for entry in rooms.values():
        if entry["last_check"] is None:
            entry["days_since"] = None
            entry["due"] = True
        else:
            entry["due"] = entry["days_since"] is None or entry["days_since"] >= 14
    return sorted(rooms.values(), key=lambda r: str(r["room"]).lower())


def _collect_video_candidates(result, after_only=False):
    """Collect .mp4 paths referenced by a result (defensive, recursive)."""
    found = []

    def walk(obj, key_hint=""):
        if isinstance(obj, dict):
            for k, v in obj.items():
                walk(v, str(k).lower())
        elif isinstance(obj, list):
            for v in obj:
                walk(v, key_hint)
        elif isinstance(obj, str):
            low = obj.lower()
            if low.endswith(".mp4"):
                if after_only and key_hint and "before" in key_hint:
                    return
                found.append(obj)

    walk(result)
    # Also any mp4 sitting in the run's outputs folder.
    if isinstance(result, dict) and result.get("run_id"):
        out_dir = ROOT / "outputs" / str(result["run_id"])
        if out_dir.is_dir():
            for p in sorted(out_dir.glob("*.mp4")):
                found.append(str(p))
    seen, out = set(), []
    for f in found:
        if f not in seen:
            seen.add(f)
            out.append(f)
    return out


def _resolve_local_path(value):
    if not isinstance(value, str) or not value:
        return None
    p = Path(value)
    if p.is_file():
        return p
    # URL-style paths like /outputs/<run>/x.mp4 or outputs/<run>/x.mp4
    rel = value.lstrip("/")
    cand = ROOT / rel
    if cand.is_file():
        return cand
    return None

def _split_csv(value: Optional[str]):
    if not value:
        return None
    items = [s.strip() for s in str(value).split(",") if s.strip()]
    return items or None


def _merge_watchlist(existing, extra):
    """Merge two watchlists, deduping case-insensitively, preserving order."""
    merged = []
    seen = set()
    for source in (existing or [], extra or []):
        for item in source:
            if not isinstance(item, str):
                continue
            cleaned = item.strip()
            if not cleaned:
                continue
            key = cleaned.lower()
            if key in seen:
                continue
            seen.add(key)
            merged.append(cleaned)
    return merged


def _json_safe(obj):
    return json.loads(json.dumps(obj, default=str))


def _new_state():
    return {"status": "running", "events": [], "cond": threading.Condition(), "result": None}


def _append_event(state, event):
    with state["cond"]:
        state["events"].append(event)
        state["cond"].notify_all()


def _delete_run_media(run_id, before, after, result, keep_media, protected=()):
    """Delete uploaded raw videos + the run's normalized videos (unless kept).

    Annotated images live in outputs/ and are never touched, so history
    (which references outputs/ images) keeps working after deletion.
    """
    if keep_media:
        log.info("run %s keep_media=1: uploaded/normalized videos kept", run_id)
        return
    protected_resolved = set()
    for item in protected or ():
        try:
            protected_resolved.add(str(Path(str(item)).resolve()))
        except Exception:
            protected_resolved.add(str(item))
    deleted = []
    # Normalized (and any other result-referenced) videos outside outputs/.
    if isinstance(result, dict):
        for cand in _collect_video_candidates(result):
            p = _resolve_local_path(cand)
            if p is None:
                continue
            try:
                resolved = str(p.resolve())
                if resolved in protected_resolved:
                    continue
                if (ROOT / "outputs") in p.resolve().parents:
                    continue
                if (ROOT / "data") in p.resolve().parents:
                    continue
                p.unlink()
                deleted.append(str(p))
            except Exception as exc:  # noqa: BLE001
                log.warning("run %s could not delete media %s: %s", run_id, cand, exc)
    # Uploaded raw videos for this server run.
    upload_dir = ROOT / "cache" / "uploads" / run_id
    try:
        if upload_dir.is_dir():
            shutil.rmtree(upload_dir, ignore_errors=True)
            deleted.append(str(upload_dir))
    except Exception as exc:  # noqa: BLE001
        log.warning("run %s could not delete uploads %s: %s", run_id, upload_dir, exc)
    # Fall back to deleting the exact input files if they live under uploads/.
    for raw in (before, after):
        try:
            p = Path(str(raw))
            if p.is_file() and (ROOT / "cache" / "uploads") in p.resolve().parents and str(p.resolve()) not in protected_resolved:
                p.unlink()
                deleted.append(str(p))
        except Exception:
            pass
    log.info("run %s media deleted after run (keep_media=0): %s", run_id, deleted or "nothing to delete")


def _run_pipeline(run_id: str, before: str, after: str, concepts, watchlist, narration: bool = False,
                  room: str = "Bedroom", filmed_by=None, keep_media: bool = False, protected_paths=()):
    state = RUNS[run_id]
    run_started = time.time()
    log.info("run %s start before=%s after=%s concepts=%s watchlist=%s narration=%s room=%s filmed_by=%s keep_media=%s",
             run_id, before, after, concepts, watchlist, narration, room, filmed_by, keep_media)

    def cb(event):
        try:
            ev = _json_safe(event) if isinstance(event, dict) else {"stage": "log", "status": "info", "message": str(event)}
        except Exception:
            ev = {"stage": "log", "status": "info", "message": str(event)}
        _append_event(state, ev)

    voice_info = None
    effective_watchlist = watchlist
    intentional_for_pipeline = None
    if narration:
        try:
            voice_started = time.time()
            transcribed = voice_mod.transcribe(before, diarize=True)
            transcript = transcribed.get("transcript", "") if isinstance(transcribed, dict) else ""
            turns = transcribed.get("turns", []) if isinstance(transcribed, dict) else []
            if not isinstance(turns, list):
                turns = []
            intent = voice_mod.extract_intent(turns if turns else transcript)
            if not isinstance(intent, dict):
                intent = {"items": [], "intentional": []}
            items = intent.get("items") or []
            intentional = intent.get("intentional") or []
            effective_watchlist = _merge_watchlist(watchlist, items) or None
            intentional_for_pipeline = intentional or None
            voice_info = {"transcript": transcript, "items": items,
                          "intentional": intentional, "turns": turns}
            _append_event(state, {
                "stage": "voice",
                "status": "done",
                "model": "muse-voice-transcribe-1.0",
                "transcript": transcript,
                "items": items,
                "intentional": intentional,
                "turns": turns,
                "elapsed_s": round(time.time() - run_started, 3),
                "cost_usd": 0.0,
            })
            log.info("run %s voice note done items=%s intentional=%s (%.2fs)",
                     run_id, items, intentional, time.time() - voice_started)
        except Exception as exc:  # noqa: BLE001
            log.exception("run %s narration failed, continuing without it: %s", run_id, exc)
            _append_event(state, {"stage": "voice", "status": "error", "message": str(exc)})

    try:
        try:
            result = pipeline.run(before, after, concepts=concepts or None, watchlist=effective_watchlist or None,
                                  on_event=cb, room=room, filmed_by=filmed_by,
                                  intentional=intentional_for_pipeline)
        except TypeError as exc:
            # Older pipeline without room/filmed_by/intentional kwargs: retry without them.
            if ("room" not in str(exc) and "filmed_by" not in str(exc)
                    and "intentional" not in str(exc) and "unexpected keyword" not in str(exc)):
                raise
            log.warning("run %s pipeline rejected room/filmed_by/intentional kwargs, retrying without: %s", run_id, exc)
            result = pipeline.run(before, after, concepts=concepts or None, watchlist=effective_watchlist or None, on_event=cb)
            if isinstance(result, dict):
                result.setdefault("room", room)
                result.setdefault("filmed_by", filmed_by)
        if result is None:
            raise RuntimeError("pipeline returned no result")
        if voice_info is not None and isinstance(result, dict):
            models = result.get("models")
            if not isinstance(models, list):
                models = [models] if models else []
                result["models"] = models
            if "muse-voice-transcribe-1.0" not in models:
                models.append("muse-voice-transcribe-1.0")
            result["narration"] = {"transcript": voice_info["transcript"], "items": voice_info["items"],
                                   "intentional": voice_info.get("intentional", []),
                                   "turns": voice_info.get("turns", [])}
        if isinstance(result, dict):
            result.setdefault("room", room)
            if filmed_by is not None:
                result.setdefault("filmed_by", filmed_by)
        safe_result = _json_safe(result)
        # Remember where the AFTER videos are, for POST /api/baseline.
        if isinstance(safe_result, dict) and safe_result.get("run_id"):
            with RUN_MEDIA_LOCK:
                RUN_MEDIA[str(safe_result["run_id"])] = {
                    "after_original": str(after),
                    "before_original": str(before),
                    "after_candidates": _collect_video_candidates(safe_result, after_only=False),
                    "result": safe_result,
                }
        # Family feed: append when a run finishes (text only).
        try:
            if isinstance(safe_result, dict):
                result_path = ROOT / "outputs" / str(safe_result.get("run_id") or "") / "result.json"
                _feed_append(_feed_entry_from_result(safe_result, result_path if result_path.is_file() else None))
        except Exception as exc:  # noqa: BLE001
            log.warning("run %s feed append failed: %s", run_id, exc)
        # Delete raw/normalized media by default; outputs/ stays for history.
        _delete_run_media(run_id, before, after, safe_result if isinstance(safe_result, dict) else {}, keep_media, protected=protected_paths)
        # Append full-result event BEFORE marking status done, so SSE clients get it.
        final_event = {"stage": "result", "status": "done", "result": safe_result}
        with state["cond"]:
            state["events"].append(final_event)
            state["cond"].notify_all()
            state["result"] = safe_result
            state["status"] = "done"
        log.info("run %s done pipeline_run_id=%s", run_id, safe_result.get("run_id") if isinstance(safe_result, dict) else "?")
    except Exception as exc:  # noqa: BLE001
        log.exception("run %s failed: %s", run_id, exc)
        _delete_run_media(run_id, before, after, {}, keep_media, protected=protected_paths)
        err_event = {"stage": "error", "status": "error", "message": str(exc)}
        with state["cond"]:
            # Avoid duplicate error events if pipeline already emitted one.
            if not any(isinstance(e, dict) and e.get("stage") == "error" for e in state["events"]):
                state["events"].append(err_event)
            state["status"] = "error"
            state["cond"].notify_all()

    if is_sample and not use_baseline:
        before_path = ROOT / "data" / "sample_before.mp4"
        after_path = ROOT / "data" / "sample_after.mp4"
        protected.extend([str(before_path), str(after_path)])
    else:

        if use_baseline and after is None and is_sample:
            after_path = ROOT / "data" / "sample_after.mp4"

            before_path = baseline_path

            if after is None:
                return JSONResponse({"error": "after file is required when using a baseline"}, status_code=400)
            before_path = baseline_path
            after_path = None  # saved below
        else:
            if before is None or after is None:
                return JSONResponse({"error": "before and after files are required unless sample=1"}, status_code=400)

            after_path = None

        if before_path is None or after_path is None:
            upload_dir = ROOT / "cache" / "uploads" / run_id
            upload_dir.mkdir(parents=True, exist_ok=True)

            async def _save(upload: UploadFile, stem: str) -> Path:

    return {"run_id": run_id}


@app.get("/api/runs/{run_id}/events")
async def api_run_events(run_id: str):
    with RUNS_LOCK:
        state = RUNS.get(run_id)
    if state is None:
        return JSONResponse({"error": "run not found"}, status_code=404)

    async def event_stream():
        idx = 0
        last_ping = time.monotonic()
        while True:
            # Poll the shared list without blocking the event loop.
            with state["cond"]:
                pending = list(state["events"][idx:])
                status = state["status"]
            sent_terminal = False
            for ev in pending:
                idx += 1
                last_ping = time.monotonic()
                yield f"data: {json.dumps(ev, default=str)}\n\n"
                if isinstance(ev, dict) and ev.get("stage") in ("result", "error"):
                    sent_terminal = True
            if sent_terminal:
                return
            # Fallback: thread finished but no terminal result/error event was produced.
            with state["cond"]:
                drained = idx >= len(state["events"])
            if status in ("done", "error") and drained and pending == []:
                return
            now = time.monotonic()
            if now - last_ping >= 15:
                last_ping = now
                yield ": ping\n\n"
            await asyncio.sleep(0.25)

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )

@app.post("/api/baseline")
async def api_set_baseline(request: Request):
    try:
        body = await request.json()
    except Exception:
        return JSONResponse({"error": "JSON body required"}, status_code=400)
    if not isinstance(body, dict):
        return JSONResponse({"error": "JSON body required"}, status_code=400)
    pipeline_run_id = str(body.get("run_id") or "").strip()
    room_name = str(body.get("room") or "Bedroom").strip() or "Bedroom"
    slug = _room_slug(room_name)
    if not slug:
        return JSONResponse({"error": "invalid room name"}, status_code=400)
    if not pipeline_run_id or "/" in pipeline_run_id or "\\" in pipeline_run_id or ".." in pipeline_run_id:
        return JSONResponse({"error": "run not found"}, status_code=404)

    result = None
    result_path = ROOT / "outputs" / pipeline_run_id / "result.json"
    if result_path.is_file():
        try:
            result = json.loads(result_path.read_text(encoding="utf-8"))
        except Exception:
            result = None
    if not isinstance(result, dict):
        # Fall back to in-memory run result.
        with RUN_MEDIA_LOCK:
            info = RUN_MEDIA.get(pipeline_run_id)
        if info and isinstance(info.get("result"), dict):
            result = info["result"]
        else:
            return JSONResponse({"error": "run not found"}, status_code=404)

    candidates = []
    with RUN_MEDIA_LOCK:
        info = RUN_MEDIA.get(pipeline_run_id) or {}
        candidates.extend(info.get("after_candidates") or [])
        if info.get("after_original"):
            candidates.append(info["after_original"])
    candidates.extend(_collect_video_candidates(result))
    # Prefer AFTER-looking / normalized candidates, then any candidate.
    ordered = sorted(candidates, key=lambda c: (0 if "after" in Path(str(c)).name.lower() else 1, 0 if "normal" in str(c).lower() else 1))
    source = None
    for cand in ordered:
        p = _resolve_local_path(cand)
        if p is not None:
            # When the hint says "before", skip obvious before files if an after exists later.
            if "before" in p.name.lower() and any("after" in Path(str(c)).name.lower() for c in ordered):
                continue
            source = p
            break
    if source is None:
        return JSONResponse({"error": "Videos were deleted after the check; film a fresh baseline."}, status_code=409)

    try:
        BASELINES_DIR.mkdir(parents=True, exist_ok=True)
        dest = BASELINES_DIR / f"{slug}.mp4"
        shutil.copyfile(str(source), str(dest))
        meta = {"room": room_name, "date": _now_iso(), "from_run": pipeline_run_id, "watchlist": _watchlist_names(result)}
        (BASELINES_DIR / f"{slug}.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")
    except Exception as exc:  # noqa: BLE001
        log.exception("baseline copy failed: %s", exc)
        return JSONResponse({"error": str(exc)}, status_code=500)
    log.info("baseline saved room=%s slug=%s from_run=%s source=%s", room_name, slug, pipeline_run_id, source)
    return _json_safe(meta)

@app.get("/api/feed/events")
async def api_feed_events():
    async def feed_stream():
        with FEED_LOCK:
            snapshot_entries = list(reversed(FEED))
            idx = len(FEED)
        yield f"data: {json.dumps({'type': 'snapshot', 'entries': snapshot_entries, 'rooms': _compute_rooms()}, default=str)}\n\n"
        last_ping = time.monotonic()
        last_due = time.monotonic()
        while True:
            await asyncio.sleep(0.5)
            # Poll the feed list index every 0.5 s (lock makes this thread-safe).
            with FEED_LOCK:
                new_entries = list(FEED[idx:])
                idx = len(FEED)
            for entry in new_entries:
                last_ping = time.monotonic()
                yield f"data: {json.dumps({'type': 'entry', 'entry': entry}, default=str)}\n\n"
            now = time.monotonic()
            if now - last_due >= 60:
                last_due = now
                last_ping = now
                yield f"data: {json.dumps({'type': 'due', 'rooms': _compute_rooms()}, default=str)}\n\n"
                continue
            if now - last_ping >= 15:
                last_ping = now
                yield ": ping\n\n"

    return StreamingResponse(
        feed_stream(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )

def _agent_context() -> dict:
    """Words-only agent context from the server's feed list + rooms data."""
    with FEED_LOCK:
        feed_snapshot = list(FEED)
    # Tolerate feed entries whose run_id is None (glasses entries): hand the
    # agent a copy keyed by the entry id so string/run_id handling in the
    # agent can never choke on None, while /api/feed keeps the original.
    safe_entries = []
    for entry in feed_snapshot:
        if isinstance(entry, dict) and entry.get("run_id") is None:
            copied = dict(entry)
            copied["run_id"] = copied.get("id") or "glasses"
            safe_entries.append(copied)
        else:
            safe_entries.append(entry)
    try:
        return agent_mod.build_context(feed_entries=safe_entries, rooms=_compute_rooms())
    except Exception:
        # Last-resort: retry with run_ids stringified, never crash /api/ask
        # just because a glasses entry is in the feed.
        sanitized = []
        for entry in safe_entries:
            if isinstance(entry, dict):
                copied = dict(entry)
                if copied.get("run_id") is None:
                    copied["run_id"] = str(copied.get("id") or "glasses")
                sanitized.append(copied)
            else:
                sanitized.append(entry)
        return agent_mod.build_context(feed_entries=sanitized, rooms=_compute_rooms())

                kept_claims = result.get("kept_claims") or result.get("claims") or []
                dropped_claims = result.get("dropped") or []
                thumb = None
                for claim in kept_claims:
                    if not isinstance(claim, dict):
                        continue
                    images = claim.get("images") or {}
                    if isinstance(images, dict) and (images.get("after") or images.get("before")):
                        thumb = images.get("after") or images.get("before")
                        break
                if not thumb:
                    keyframes = result.get("keyframes") or {}
                    if isinstance(keyframes, dict):
                        for side in ("after", "before"):
                            urls = keyframes.get(side) or []
                            if urls:
                                thumb = urls[0]
                                break
                entries.append(
                    {
                        "run_id": result.get("run_id") or path.parent.name,
                        "summary": result.get("summary") or "",
                        "kept": len(kept_claims),
                        "dropped": len(dropped_claims),
                        "cost_usd": result.get("cost_usd"),
                        "latency_s": result.get("latency_s"),
                        "thumb": thumb,
                        "room": result.get("room"),
                        "date": _result_datetime(result, path).isoformat(),
                    }
                )
            except Exception:
                continue
    return _json_safe(entries)

def _glasses_alias_map(room_name: str) -> dict:
    """{lowercase watchlist item: alias} from the newest result.json for room.

    Looks at the newest outputs/*/result.json whose room matches (falling
    back to no aliases). Shape: result['aliases'][item] is either
    {'alias': str, ...} or a plain string.
    """
    outputs_dir = ROOT / "outputs"
    if not outputs_dir.is_dir():
        return {}
    try:
        files = sorted(outputs_dir.glob("*/result.json"),
                       key=lambda p: p.stat().st_mtime if p.exists() else 0, reverse=True)
    except Exception:
        return {}
    target_slug = _room_slug(room_name) or "bedroom"
    for path in files:
        try:
            result = json.loads(path.read_text(encoding="utf-8"))
        except Exception:
            continue
        if not isinstance(result, dict):
            continue
        result_room = str(result.get("room") or "Bedroom")
        if _room_slug(result_room) != target_slug and result_room.strip().lower() != str(room_name).strip().lower():
            continue
        # This is the newest result for the room; use its aliases (if any).
        aliases = result.get("aliases")
        if not isinstance(aliases, dict):
            return {}
        out = {}
        for key, val in aliases.items():
            if not isinstance(key, str) or not key.strip():
                continue
            alias = None
            if isinstance(val, dict):
                alias = val.get("alias")
            elif isinstance(val, str):
                alias = val
            if isinstance(alias, str) and alias.strip():
                out[key.strip().lower()] = alias.strip()
        return out
    return {}


def _glasses_direction_distance(box, w, h):
    try:
        x1, y1, x2, y2 = [float(v) for v in (list(box) + [0, 0, 0, 0])[:4]]
    except Exception:
        return "ahead", "across the room"
    try:
        width = float(w) if w else 0.0
    except Exception:
        width = 0.0
    try:
        height = float(h) if h else 0.0
    except Exception:
        height = 0.0
    if width > 0:
        frac_x = ((x1 + x2) / 2.0) / width
    else:
        frac_x = 0.5
    if frac_x < 1.0 / 3.0:
        direction = "on your left"
    elif frac_x < 2.0 / 3.0:
        direction = "ahead"
    else:
        direction = "on your right"
    if height > 0:
        frac_h = (y2 - y1) / height
    else:
        frac_h = 0.0
    if frac_h > 0.55:
        distance = "close"
    elif frac_h > 0.25:
        distance = "a few steps away"
    else:
        distance = "across the room"
    return direction, distance


def _glasses_best_detection(parsed):
    """Flatten {frame: [dets]} (or a bare list) and return the highest-c det."""
    best = None
    if isinstance(parsed, dict):
        iterable = []
        for val in parsed.values():
            if isinstance(val, list):
                iterable.extend(val)
            elif isinstance(val, dict):
                iterable.append(val)
    elif isinstance(parsed, list):
        iterable = parsed
    else:
        iterable = []
    for det in iterable:
        if not isinstance(det, dict) or not isinstance(det.get("box"), (list, tuple)):
            continue
        try:
            c = float(det.get("c", 0.0))
        except Exception:
            continue
        if best is None or c > float(best.get("c", 0.0)):
            best = det
    return best


def _glasses_frame_sync(image_bytes: bytes, filename: str, content_type: str, items_with_alias):
    """Run SAM per watchlist item concurrently; frame stays in memory.

    Only if the in-memory (ndarray) path fails / is unavailable do we write
    a NamedTemporaryFile, and it is deleted immediately in a finally block.
    Frames are never stored under cache/, outputs/ or data/.
    """
    suffix = Path(filename or "").suffix.lower()
    if suffix not in (".jpg", ".jpeg", ".png"):
        suffix = ".png" if content_type and "png" in str(content_type).lower() else ".jpg"

    # Decode in memory if possible.
    img_input = None
    try:
        import cv2  # local import: server still starts without cv2
        import numpy as np
        arr = cv2.imdecode(np.frombuffer(image_bytes, dtype=np.uint8), cv2.IMREAD_COLOR)
        if arr is not None:
            img_input = arr
    except Exception as exc:  # noqa: BLE001
        log.warning("glasses frame in-memory decode unavailable: %s", exc)
        img_input = None

    tmp_path = None
    if img_input is None:
        # sam.segment_image needs a path in this environment.
        fh = tempfile.NamedTemporaryFile(delete=False, suffix=suffix)
        tmp_path = fh.name
        try:
            fh.write(image_bytes)
        finally:
            fh.close()
        img_input = tmp_path

    def _segment_one(pair):
        concept, alias = pair
        try:
            parsed = sam_mod.segment_image(img_input, alias)
        except Exception as exc:  # noqa: BLE001
            # Retry once via a per-item temp file if the ndarray path failed.
            if tmp_path is not None:
                log.warning("glasses segment failed concept=%s alias=%s: %s", concept, alias, exc)
                return concept, alias, None
            log.warning("glasses segment ndarray failed concept=%s, retrying via temp file: %s", concept, exc)
            retry_tmp = None
            try:
                fh = tempfile.NamedTemporaryFile(delete=False, suffix=suffix)
                retry_tmp = fh.name
                try:
                    fh.write(image_bytes)
                finally:
                    fh.close()
                parsed = sam_mod.segment_image(retry_tmp, alias)
            except Exception as exc2:  # noqa: BLE001
                log.warning("glasses segment failed concept=%s alias=%s: %s", concept, alias, exc2)
                return concept, alias, None
            finally:
                if retry_tmp:
                    try:
                        os.unlink(retry_tmp)
                    except Exception:
                        pass
        return concept, alias, _glasses_best_detection(parsed)

    try:
        if items_with_alias:
            with ThreadPoolExecutor(max_workers=min(6, len(items_with_alias))) as pool:
                results = list(pool.map(_segment_one, items_with_alias))
        else:
            results = []
    finally:
        if tmp_path:
            try:
                os.unlink(tmp_path)
            except Exception:
                pass
    return results


@app.post("/api/glasses/frame")
async def api_glasses_frame(
    image: UploadFile = File(...),
    watchlist: Optional[str] = Form(None),
    room: Optional[str] = Form(None),
):
    started = time.time()
    room_name = (str(room).strip() if room and str(room).strip() else "Bedroom")
    # Watchlist: comma string, max 4 items, deduped case-insensitively.
    raw_items = _split_csv(watchlist) or []
    seen, items = set(), []
    for item in raw_items:
        key = item.lower()
        if key not in seen:
            seen.add(key)
            items.append(item)
    items = items[:4]
    try:
        image_bytes = await image.read()
    except Exception as exc:  # noqa: BLE001
        return JSONResponse({"error": str(exc)}, status_code=400)
    if not image_bytes:
        return JSONResponse({"error": "image is required"}, status_code=400)
    if not items:
        return _json_safe({"detections": [], "whisper": "", "latency_s": round(time.time() - started, 3)})

    alias_map = _glasses_alias_map(room_name)
    items_with_alias = [(item, alias_map.get(item.strip().lower(), item)) for item in items]
    log.info("glasses frame room=%s items=%s aliases=%s bytes=%d", room_name, items, items_with_alias, len(image_bytes))
    try:
        results = await asyncio.to_thread(
            _glasses_frame_sync, image_bytes, image.filename or "", image.content_type or "", items_with_alias)
    except Exception as exc:  # noqa: BLE001
        log.exception("glasses frame failed: %s", exc)
        return JSONResponse({"error": str(exc)}, status_code=502)

    detections = []
    whisper_parts = []
    for concept, alias, best in results:
        if best is None:
            continue
        try:
            c = float(best.get("c", 0.0))
        except Exception:
            continue
        if c < 0.45:
            continue
        box = best.get("box")
        try:
            box_out = [int(float(v)) for v in list(box)[:4]]
        except Exception:
            box_out = list(box)  # type: ignore[arg-type]
        try:
            w_out = int(float(best.get("w", 0)))
        except Exception:
            w_out = best.get("w")
        try:
            h_out = int(float(best.get("h", 0)))
        except Exception:
            h_out = best.get("h")
        direction, distance = _glasses_direction_distance(box_out, w_out, h_out)
        detections.append({
            "concept": concept,
            "alias": alias,
            "box": box_out,
            "c": c,
            "w": w_out,
            "h": h_out,
            "direction": direction,
            "distance": distance,
        })
        display = concept.strip()
        display = (display[:1].upper() + display[1:]) if display else concept
        whisper_parts.append(f"{display} — {direction}, {distance}.")
    whisper = " ".join(whisper_parts)
    latency_s = round(time.time() - started, 3)
    log.info("glasses frame done detections=%d latency_s=%s", len(detections), latency_s)
    return _json_safe({"detections": detections, "whisper": whisper, "latency_s": latency_s})
