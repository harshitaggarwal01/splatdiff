"""Muse Wonder 1.0 escalation and trend analysis.

Wonder is used only where the spec calls for it:
- as an escalation judge when Muse Spark's moved verdict is uncertain
  (confidence in [0.55, 0.80)), accepting the same ``input_image`` parts
  and request shape as Spark via ``POST /v1/responses``;
- as a text-only trend analyst across earlier family-feed entries for
  the same room.

All calls go through :mod:`splatdiff.meta_api`, so retries, the disk
response cache and the cost ledger behave exactly as for Spark.
"""
from __future__ import annotations

MODEL = "muse-wonder-1.0"


def _make_body(content: list, max_tokens: int = 2000, retry: bool = False) -> dict:
    body = {"model": MODEL, "input": [user_message(content)],
            "reasoning": {"effort": "low"}, "max_output_tokens": max_tokens}
    if retry:
        # Changed request hash so a cached bad/empty first response is
        # not served again for the retry.
        body["metadata"] = {"retry": "1"}
    return body


def _call_json(content: list, max_tokens: int = 2000, what: str = "wonder"):
    """Call Wonder and parse strict JSON; retry once, return None on failure."""
    last_exc: Exception | None = None
    for attempt in (0, 1):
        retry = attempt == 1
        try:
            text = extract_text(responses(_make_body(content, max_tokens=max_tokens, retry=retry)))
        except Exception as e:  # noqa: BLE001
            last_exc = e
            log.warning("%s: Wonder call failed (attempt %d/2): %s", what, attempt + 1, e)
            continue
        if not text or not str(text).strip():
            last_exc = ValueError("empty text, cannot parse JSON")
            log.warning("%s: empty text from Wonder (attempt %d/2)", what, attempt + 1)
            continue
        try:
            return parse_json_loose(text)
        except Exception as e:  # noqa: BLE001
            last_exc = e
            log.warning("%s parse failed (attempt %d/2): %s; text=%s",
                        what, attempt + 1, e, str(text)[:500])
            continue
    log.warning("%s: giving up after retries: %s", what, last_exc)
    return None


def _crop(frame_path, box, out_path):
    img = cv2.imread(str(frame_path))
    if img is None or not box:
        return None
    try:
        h, w = img.shape[:2]
        x1, y1, x2, y2 = [int(v) for v in box]
    except Exception:  # noqa: BLE001
        return None
    pad_x, pad_y = int(0.6 * max(1, x2 - x1)), int(0.6 * max(1, y2 - y1))
    crop = img[max(0, y1 - pad_y):min(h, y2 + pad_y), max(0, x1 - pad_x):min(w, x2 + pad_x)]
    if crop.size == 0:
        return None
    cv2.imwrite(str(out_path), crop)
    return str(out_path)


def judge_moved_wonder(concept: str, before_img, after_img, before_box=None, after_box=None) -> dict | None:
    """Escalation moved-judgement with muse-wonder-1.0.

    Same question and same before/after evidence images (SAM box drawn,
    plus context crops) as :func:`splatdiff.spark.judge_moved`.
    Returns {"moved": bool, "confidence": float, "reason": str} or
    None on any failure, so the caller can keep the Spark verdict.
    """
    try:
        tmp = Path("cache") / "judge_wonder"
        tmp.mkdir(parents=True, exist_ok=True)
        safe = "".join(c if c.isalnum() else "_" for c in str(concept).lower())
        bp, ap = tmp / f"{safe}_before.jpg", tmp / f"{safe}_after.jpg"
        cv2.imwrite(str(bp), draw_box(before_img, before_box, concept, (0, 165, 255)))
        cv2.imwrite(str(ap), draw_box(after_img, after_box, concept, (0, 165, 255)))
        content = [text_part(
            f"The orange box marks '{concept}' in the BEFORE frame and the AFTER frame of the same room. "
            "Camera viewpoint differs between videos, so do NOT judge by raw pixel position. Judge position relative to "
            "surrounding furniture and room landmarks. Also shown are context crops. Return STRICT JSON only: "
            '{"moved": true, "confidence": 0.0, "reason": "short reason"} with confidence in 0..1.'),
            text_part("BEFORE full frame"), image_part(str(bp)),
            text_part("AFTER full frame"), image_part(str(ap))]
        for label, src, box, name in (("BEFORE crop", str(bp), before_box, "b_crop"),
                                      ("AFTER crop", str(ap), after_box, "a_crop")):
            crop_path = _crop(src, box, tmp / f"{safe}_{name}.jpg")
            if crop_path:
                content.append(text_part(label))
                content.append(image_part(crop_path))
        d = _call_json(content, max_tokens=2000, what=f"judge_moved_wonder {concept}")
        if not isinstance(d, dict):
            return None
        try:
            conf = float(d.get("confidence", 0.0) or 0.0)
        except Exception:  # noqa: BLE001
            conf = 0.0
        return {"moved": bool(d.get("moved")),
                "confidence": max(0.0, min(1.0, conf)),
                "reason": str(d.get("reason", "") or "")}
    except Exception as e:  # noqa: BLE001
        log.warning("judge_moved_wonder failed for %s: %s", concept, e)
        return None


def analyse_trend(feed_entries, kept_claims, watchlist_status) -> dict | None:
    """Text-only trend across earlier visits, with muse-wonder-1.0.

    ``feed_entries`` are earlier family-feed entries for the same room,
    newest first (max 20, enforced by the caller and re-enforced here).
    Returns {"trend": str|None, "repeat_items": [{"concept", "times",
    "of_checks"}]} or None on any failure (caller skips silently).
    """
    try:
        entries = [e for e in (feed_entries or []) if isinstance(e, dict)][:20]
        payload = {"earlier_checks_newest_first": entries,
                   "this_run_kept_claims": kept_claims or [],
                   "this_run_watchlist_status": watchlist_status or []}
        content = [text_part(
            "You are summarising the trend across repeated 'Second Look' room checks for an elderly parent's room, "
            "for a family caregiver. Below are earlier family-feed entries for the SAME room (newest first, at most "
            "20), plus this run's kept (SAM-grounded) claims and watchlist status. Look for items that keep changing "
            "or keep being flagged across checks. Return STRICT JSON only, no markdown: "
            '{"trend": "one plain sentence or null", "repeat_items": [{"concept": "str", "times": 0, "of_checks": 0}]} '
            "where 'trend' is ONE plain sentence describing the overall trend across visits (or null if there is no "
            "meaningful pattern), and 'repeat_items' lists concepts flagged in at least 2 checks, with 'times' = in "
            "how many of those checks the concept was flagged and 'of_checks' = the total number of checks considered "
            "(earlier entries plus this run). Do not invent items that never appear in the data.\nDATA:\n"
            + json.dumps(payload, default=str))]
        d = _call_json(content, max_tokens=2000, what="analyse_trend")
        if not isinstance(d, dict):
            return None
        trend = d.get("trend")
        if trend is not None:
            trend = str(trend).strip() or None
        repeat_items = []
        raw_items = d.get("repeat_items") or []
        if isinstance(raw_items, (list, tuple)):
            for item in raw_items:
                if not isinstance(item, dict):
                    continue
                concept = " ".join(str(item.get("concept", "") or "").strip().lower().split())
                if not concept:
                    continue
                try:
                    times = int(item.get("times", 0) or 0)
                except Exception:  # noqa: BLE001
                    times = 0
                try:
                    of_checks = int(item.get("of_checks", 0) or 0)
                except Exception:  # noqa: BLE001
                    of_checks = 0
                repeat_items.append({"concept": concept, "times": times, "of_checks": of_checks})
        return {"trend": trend, "repeat_items": repeat_items}
    except Exception as e:  # noqa: BLE001
        log.warning("analyse_trend failed: %s", e)
        return None

def _safe(s: str) -> str:
    return "".join(c if c.isalnum() else "_" for c in str(s).lower()).strip("_")

def _norm_concept(s: str) -> str:
    return " ".join(str(s).strip().lower().split())

def _clean_list(items) -> list[str]:
    """Accept None / comma string / iterable; return stripped strings."""
    if not items:
        return []
    if isinstance(items, str):
        items = [c.strip() for c in items.split(",")]
    out = []
    for item in items:
        c = str(item).strip()
        if c:
            out.append(c)
    return out

def _clean_intentional(intentional) -> list[dict]:
    """Normalise the ``intentional`` kwarg to [{'concept','quote','speaker'}]."""
    out: list[dict] = []
    if not intentional:
        return out
    if isinstance(intentional, dict):
        intentional = intentional.get("intentional") or intentional.get("items") or []
    if isinstance(intentional, str):
        intentional = [c.strip() for c in intentional.split(",") if c.strip()]
    for item in intentional or []:
        if isinstance(item, dict):
            concept = " ".join(str(item.get("concept", "") or "").strip().lower().split())
            if not concept:
                continue
            out.append({"concept": concept,
                        "quote": str(item.get("quote", "") or ""),
                        "speaker": str(item.get("speaker", "") or "unknown") or "unknown"})
        elif isinstance(item, str) and item.strip():
            out.append({"concept": " ".join(item.strip().lower().split()), "quote": "", "speaker": "unknown"})
    return out

def _token_overlap(a: str, b: str) -> int:
    return len(set(_norm_concept(a).split()) & set(_norm_concept(b).split()))

def _match_intentional(concept: str, intentional_list: list[dict]):
    """Best token-overlap intentional item for a concept, or None."""
    best, best_overlap = None, 0
    for item in intentional_list or []:
        overlap = _token_overlap(concept, str(item.get("concept", "")))
        if overlap > best_overlap:
            best, best_overlap = item, overlap
    return best if best_overlap > 0 else None

def _dedupe_prepend(priority: list[str], rest: list[str], cap: int = MAX_CONCEPTS) -> list[str]:
    """priority first, then rest; dedupe case-insensitively; cap total."""
    out, seen = [], set()
    for c in list(priority) + list(rest):
        key = _norm_concept(c)
        if not key or key in seen:
            continue
        seen.add(key); out.append(str(c).strip())
        if len(out) >= cap:
            break
    return out

def _find_summary(concept: str, summaries: dict):
    """Exact (normalised) match, else best token-overlap match."""
    n = _norm_concept(concept)
    for k, v in summaries.items():
        if _norm_concept(k) == n:
            return k, v
    best_key, best_v, best_overlap = None, None, 0
    for k, v in summaries.items():
        overlap = len(set(_norm_concept(k).split()) & set(n.split()))
        if overlap > best_overlap:
            best_key, best_v, best_overlap = k, v, overlap
    return best_key, best_v

def _run_vocab_repair(summaries: dict, watchlist_list: list, before_norm, after_norm,
                      before_keyframes: list, after_keyframes: list, prog, emit=None,
                      before_tracks: dict | None = None, after_tracks: dict | None = None) -> dict:
    def _emit(ev: dict):
        if emit is not None:
            try:
                emit(ev)
            except Exception:
                log.exception("on_event callback failed for %s", ev)

    empty = {"aliases": {}, "repair_sam_calls": 0, "repaired": 0,
             "spark_calls": 0, "sam_requests": 0, "sam_frames": 0}
    watch_norm = [_norm_concept(w) for w in (watchlist_list or [])]

    def _priority(concept: str) -> int:
        n = _norm_concept(concept)
        return watch_norm.index(n) if n in watch_norm else len(watch_norm)

    candidates = [c for c, s in summaries.items() if isinstance(s, dict) and s.get("label") == "not_found"]
    candidates = sorted(candidates, key=_priority)[:MAX_REPAIR_CONCEPTS]
    if not candidates:
        return empty

    _emit({"stage": "repair", "status": "start", "concepts": list(candidates)})
    prog(f"Vocabulary repair: SAM found nothing for {len(candidates)} concept(s) "
         f"({', '.join(candidates)}); trying Spark-suggested aliases…")
    spark_keyframes = list(before_keyframes or [])[:2] + list(after_keyframes or [])[:2]
    test_keyframes = list(before_keyframes or [])[:3] + list(after_keyframes or [])[:3]

    aliases_result: dict = {}
    repair_sam_calls = 0
    repaired = 0
    spark_calls = 0
    retrack_requests = 0
    retrack_frames = 0

    for concept in candidates:
        tried: list[str] = []
        best_alias = None
        best_c = 0.0
        spark_calls += 1
        try:
            raw_aliases = suggest_aliases(concept, spark_keyframes)
        except Exception as e:
            log.warning("suggest_aliases failed for %s: %s", concept, e)
            raw_aliases = []
        seen = set()
        for a in raw_aliases or []:
            alias = " ".join(str(a or "").strip().lower().split())
            if alias and alias != _norm_concept(concept) and alias not in seen:
                seen.add(alias); tried.append(alias)
        tried = tried[:3]

        if tried and test_keyframes:
            tasks = [(alias, kf) for alias in tried for kf in test_keyframes]
            per_alias_best: dict[str, float] = {}
            repair_sam_calls += len(tasks)

            def _test(task):
                alias, kf = task
                try:
                    parsed = segment_image(kf, alias)
                except Exception as e:
                    log.warning("SAM alias test failed for %s as '%s' on %s: %s", concept, alias, kf, e)
                    return alias, None
                return alias, _best_detection(parsed)

            with ThreadPoolExecutor(max_workers=6) as ex:
                futures = [ex.submit(_test, t) for t in tasks]
                for fut in as_completed(futures):
                    try:
                        alias, det = fut.result()
                    except Exception:
                        log.exception("SAM alias test job failed for %s", concept)
                        continue
                    if det is not None and float(det["c"]) > per_alias_best.get(alias, 0.0):
                        per_alias_best[alias] = float(det["c"])
            for alias in tried:
                cval = per_alias_best.get(alias, 0.0)
                if cval >= REPAIR_MIN_CONF and cval > best_c:
                    best_alias, best_c = alias, float(cval)

        if best_alias is not None:
            alias_before, alias_after = {}, {}
            try:
                alias_before = track_concept(before_norm, best_alias) or {}
            except Exception as e:
                log.exception("SAM repair tracking (before) failed for %s as '%s': %s", concept, best_alias, e)
            try:
                alias_after = track_concept(after_norm, best_alias) or {}
            except Exception as e:
                log.exception("SAM repair tracking (after) failed for %s as '%s': %s", concept, best_alias, e)
            repair_sam_calls += 2
            retrack_requests += 2
            try:
                retrack_frames += len(alias_before) + len(alias_after)
            except Exception:
                pass
            norm_before = {int(k): v for k, v in dict(alias_before).items()}
            norm_after = {int(k): v for k, v in dict(alias_after).items()}
            if before_tracks is not None:
                before_tracks[concept] = norm_before
            if after_tracks is not None:
                after_tracks[concept] = norm_after
            try:
                recomputed = diffmod.summarize_tracks({concept: norm_before}, {concept: norm_after})
                if concept in recomputed:
                    summaries[concept] = recomputed[concept]
            except Exception:
                log.exception("summarize_tracks failed for repaired concept %s", concept)
            if isinstance(summaries.get(concept), dict):
                summaries[concept]["alias"] = best_alias
            aliases_result[concept] = {"alias": best_alias, "tried": list(tried), "c": float(best_c)}
            repaired += 1
            prog(f"Repair: '{concept}' tracked as \"{best_alias}\" (best image c={best_c:.2f}); "
                 f"label now '{summaries.get(concept, {}).get('label', 'not_found')}'")
        else:
            prog(f"Repair: no working alias for '{concept}' (tried: {', '.join(tried) or 'none'})")
        _emit({"stage": "repair", "status": "progress", "concept": concept,
               "alias": best_alias, "tried": list(tried),
               "c": float(best_c) if best_alias is not None else 0.0})

    _emit({"stage": "repair", "status": "done", "repaired": int(repaired)})
    return {"aliases": aliases_result, "repair_sam_calls": int(repair_sam_calls),
            "repaired": int(repaired), "spark_calls": int(spark_calls),
            "sam_requests": int(retrack_requests), "sam_frames": int(retrack_frames)}

def _run_image_rechecks(summaries: dict, before_keyframes: list, after_keyframes: list, prog,
                        emit=None) -> int:
    def _emit(ev: dict):
        if emit is not None:
            try:
                emit(ev)
            except Exception:
                log.exception("on_event callback failed for %s", ev)

    candidates = [(c, s) for c, s in summaries.items() if s.get("label") in ("added", "removed")]
    _emit({"stage": "recheck", "status": "start", "concept": "", "found": False, "c": 0.0})
    if not candidates:
        _emit({"stage": "recheck", "status": "done", "concept": "", "found": False, "c": 0.0})
        return 0
    prog(f"SAM image re-check for {len(candidates)} added/removed concept(s)…")
    absent_side = {}
    for concept, s in candidates:
        absent_side[concept] = "before" if s["label"] == "added" else "after"

    def _job(item):
        concept, s = item
        side = absent_side[concept]
        keyframes = before_keyframes if side == "before" else after_keyframes
        best, calls = None, 0
        for idx, kf_path in enumerate(keyframes or []):
            calls += 1
            try:
                parsed = segment_image(kf_path, concept)
            except Exception as e:
                log.warning("SAM image re-check failed for %s keyframe %s: %s", concept, idx, e)
                continue
            det = _best_detection(parsed)
            if det is not None and (best is None or det["c"] > best["c"]):
                best = {"video": side, "keyframe_index": idx, "box": det["box"], "c": det["c"]}
        return concept, side, best, calls

    found: dict[str, dict] = {}
    total_calls = 0
    with ThreadPoolExecutor(max_workers=6) as ex:
        futures = {ex.submit(_job, item): item[0] for item in candidates}
        for fut in as_completed(futures):
            concept = futures[fut]
            try:
                concept, side, det, calls = fut.result()
            except Exception as e:
                log.exception("SAM image re-check job failed for %s", concept)
                det, calls = None, 0
            total_calls += calls
            if det is not None:
                found[concept] = det
            _emit({"stage": "recheck", "status": "progress", "concept": concept,
                   "found": det is not None,
                   "c": float(det["c"]) if det is not None else 0.0})
    for concept, s in candidates:
        side = absent_side[concept]
        keyframes = before_keyframes if side == "before" else after_keyframes
        if concept in found:
            det = found[concept]
            diffmod.apply_image_recheck(summaries, concept,
                                        {"checked": True, "found": True, "video": side,
                                         "keyframe_index": det["keyframe_index"],
                                         "box": det["box"], "c": det["c"]})
            prog(f"Re-check: SAM image found '{concept}' in {side.upper()} "
                 f"keyframe {det['keyframe_index']} (c={det['c']:.2f}) - label now 'present_both' (was '{s.get('original_label')}')")
        else:
            diffmod.apply_image_recheck(summaries, concept,
                                        {"checked": True, "found": False, "video": side,
                                         "keyframe_index": None, "box": None, "c": 0.0,
                                         "keyframes_checked": len(keyframes or [])})
    _emit({"stage": "recheck", "status": "done", "concept": "", "found": False, "c": 0.0})
    return total_calls

def _judge_frame(side: str, s: dict, norm_path, keyframes: list):
    recheck = s.get("recheck") or {}
    if recheck.get("found") and str(recheck.get("video", "")).lower() == side:
        idx = recheck.get("keyframe_index")
        if idx is not None and keyframes and 0 <= int(idx) < len(keyframes):
            frame = cv2.imread(str(keyframes[int(idx)]))
            if frame is not None:
                return frame, recheck.get("box")
        log.warning("re-check keyframe unavailable for %s side %s; falling back to video frame",
                    s.get("concept"), side)
    summ = s[side]
    frame = read_frame(norm_path, summ["best_frame"] or 0) if summ["best_frame"] is not None else read_frame(norm_path, 0)
    return frame, summ["best_box"]

def _watchlist_note(concept: str, status: str) -> str:
    c = str(concept)
    if status == "missing":
        return (f"{c} is no longer where it was - check it is within reach of the bed/chair "
                f"and find it before it is needed.")
    if status == "moved":
        return (f"{c} has moved from where it was - check it is within reach of the bed/chair "
                f"and has not become a trip hazard.")
    if status == "not_seen":
        return (f"{c} was not detected in either video - check it manually; SAM may have missed it, "
                f"or it may have been removed before the BEFORE video.")
    if status == "appeared":
        return (f"{c} was not seen in the BEFORE video but appears in the AFTER video - check it is "
                f"stored safely, within reach if it is needed, and not blocking a walkway.")
    return f"{c} appears to still be in place."

_WATCHLIST_RISK = {"missing": "high", "moved": "high", "not_seen": "medium",
                   "appeared": "medium", "in_place": "low"}

def _build_watchlist_status(watchlist_list: list[str], summaries: dict, kept: list,
                            moved_verdicts: dict, images_for) -> list[dict]:
    moved_kept = {_norm_concept(c.get("concept", "")) for c in kept if c.get("change") == "moved"}
    out = []
    for item in watchlist_list:
        key, s = _find_summary(item, summaries)
        concept = key or _norm_concept(item)
        if s is None:
            status = "not_seen"
        else:
            label = s.get("label", "not_found")
            if label == "removed":
                status = "missing"
            elif label == "added":
                status = "appeared"
            elif label == "present_both":
                v = moved_verdicts.get(key) or {}
                try:
                    v_conf = float(v.get("confidence", 0.0) or 0.0)
                except Exception:
                    v_conf = 0.0
                verdict_kept = bool(v.get("moved")) and v_conf >= diffmod.MOVED_MIN_CONF
                status = "moved" if (_norm_concept(key) in moved_kept or verdict_kept) else "in_place"
            else:
                status = "not_seen"
        note = _watchlist_note(concept, status)
        alias = s.get("alias") if isinstance(s, dict) else None
        if alias and f'tracked as "{alias}"' not in note:
            note = f"{note} (tracked as \"{alias}\")"
        out.append({"concept": concept, "status": status,
                    "risk": _WATCHLIST_RISK.get(status, "medium"),
                    "note": note,
                    "images": images_for(concept)})
    return out

def _read_feed_entries(room: str) -> list[dict]:
    """Earlier family-feed entries for the same room, in file order."""
    entries: list[dict] = []
    try:
        if not FEED_PATH.is_file():
            return entries
        for line in FEED_PATH.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except Exception:
                continue
            if isinstance(obj, dict) and _norm_concept(obj.get("room", "")) == _norm_concept(room):
                entries.append(obj)
    except Exception as e:  # noqa: BLE001
        log.warning("could not read feed for trend: %s", e)
        return []
    return entries

def run(before, after, concepts=None, out_dir="outputs", progress=None, on_event=None, watchlist=None,
        room=None, filmed_by=None, intentional=None) -> dict:
    def prog(msg: str):
        log.info(msg)
        if progress:
            progress(msg)

    start, cost_start = time.time(), total_cost()
    run_id = datetime.now().strftime("%Y%m%d_%H%M%S")
    room = str(room).strip() if room and str(room).strip() else "Bedroom"
    filmed_by = str(filmed_by).strip() if filmed_by and str(filmed_by).strip() else None
    watchlist_list = _clean_list(watchlist)
    intentional_list = _clean_intentional(intentional)
    event_lock = threading.Lock()
    # Mutable counters for stages that also run inside annotation helpers.
    counters = {"privacy_checks": 0, "coverage_calls": 0, "floor_checks": 0, "floor_spark_calls": 0}
    wonder_state = {"called": False}
    wonder_lock = threading.Lock()

    def _mark_wonder_called():
        with wonder_lock:
            wonder_state["called"] = True

    def _running_cost() -> float:
        try:
            return round(float(total_cost()) - float(cost_start), 6)
        except Exception:
            return 0.0

    def emit(ev: dict):
        """Thread-safe structured event: adds elapsed_s + running cost_usd."""
        if on_event is None:
            return
        event = dict(ev)
        event["elapsed_s"] = round(time.time() - start, 3)
        event["cost_usd"] = _running_cost()
        with event_lock:
            try:
                on_event(event)
            except Exception:
                log.exception("on_event callback failed for %s", event)

    try:
        out = Path(out_dir) / run_id; out.mkdir(parents=True, exist_ok=True)

        def _url(filename: str) -> str:
            return f"/outputs/{run_id}/{filename}"

        def _images_for(concept: str) -> dict:
            safe = _safe(concept)
            b_name, a_name = f"before_{safe}.jpg", f"after_{safe}.jpg"
            return {"before": _url(b_name) if (out / b_name).exists() else None,
                    "after": _url(a_name) if (out / a_name).exists() else None}

        def _redact_frame_arr(frame):
            """Privacy-blur a raw ndarray frame (annotated evidence).

            Runs the same two SAM image checks as the keyframe privacy
            stage and blurs any sensitive boxes. Never raises.
            """
            if frame is None:
                return frame
            counters["privacy_checks"] += 2
            try:
                sensitive = detect_sensitive(frame)
                boxes = (sensitive or {}).get("boxes") or []
                if boxes:
                    return redact_image(frame, boxes)
            except Exception as e:
                log.warning("annotated-frame redaction failed: %s", e)
            return frame

        # 1. Normalize
        emit({"stage": "normalize", "status": "start"})
        prog("Normalising videos (ffmpeg, 5 fps, max 30 s)…")
        before_norm, after_norm = normalize_video(before), normalize_video(after)
        emit({"stage": "normalize", "status": "done"})

        # 2. Keyframes
        prog("Extracting keyframes…")
        bf = extract_keyframes(before_norm, 6, out / "keyframes_before")
        af = extract_keyframes(after_norm, 6, out / "keyframes_after")
        keyframe_urls = {
            "before": [f"/outputs/{run_id}/keyframes_before/{Path(p).name}" for p in (bf or [])],
            "after": [f"/outputs/{run_id}/keyframes_after/{Path(p).name}" for p in (af or [])],
        }
        emit({"stage": "keyframes", "status": "done",
              "before": list(keyframe_urls["before"]), "after": list(keyframe_urls["after"])})

        # 2b. Privacy — BEFORE any keyframe is sent to Muse Spark.
        # SAM 3.1 itself sees raw frames because it is the detector.
        privacy = {"people_blurred": 0, "documents_blurred": 0, "frames_checked": 0, "note": PRIVACY_NOTE}
        emit({"stage": "privacy", "status": "start"})
        prog("Blurring people and paper documents in keyframes (SAM 3.1)…")
        try:
            all_kf = [p for p in list(bf or []) + list(af or []) if p is not None]
            privacy["frames_checked"] = int(len(all_kf))
            counters["privacy_checks"] += 2 * len(all_kf)

            def _privacy_job(kf):
                people, docs = [], []
                try:
                    sensitive = detect_sensitive(kf) or {}
                    people = list(sensitive.get("person") or [])
                    docs = list(sensitive.get("paper document") or sensitive.get("paper_document") or [])
                    boxes = list(sensitive.get("boxes") or (people + docs))
                    if boxes:
                        redact_image(kf, boxes)  # in place for path inputs
                except Exception as e:
                    log.warning("privacy redaction failed for keyframe %s: %s", kf, e)
                return len(people), len(docs)

            if all_kf:
                with ThreadPoolExecutor(max_workers=6) as ex:
                    futures = [ex.submit(_privacy_job, kf) for kf in all_kf]
                    for fut in as_completed(futures):
                        try:
                            n_people, n_docs = fut.result()
                        except Exception:
                            log.exception("privacy job failed")
                            n_people, n_docs = 0, 0
                        privacy["people_blurred"] += int(n_people)
                        privacy["documents_blurred"] += int(n_docs)
            emit({"stage": "privacy", "status": "done",
                  "people_blurred": int(privacy["people_blurred"]),
                  "documents_blurred": int(privacy["documents_blurred"]),
                  "frames_checked": int(privacy["frames_checked"])})
            prog(f"Privacy: blurred {privacy['people_blurred']} people, "
                 f"{privacy['documents_blurred']} documents across {privacy['frames_checked']} keyframes.")
        except Exception as e:
            log.exception("privacy stage failed; continuing")
            privacy = {"people_blurred": int(privacy.get("people_blurred", 0) or 0),
                       "documents_blurred": int(privacy.get("documents_blurred", 0) or 0),
                       "frames_checked": int(privacy.get("frames_checked", 0) or 0),
                       "note": PRIVACY_NOTE}
            emit({"stage": "privacy", "status": "done",
                  "people_blurred": privacy["people_blurred"],
                  "documents_blurred": privacy["documents_blurred"],
                  "frames_checked": privacy["frames_checked"], "error": str(e)})

        # 3. Inventory (watchlist always tracked: prepended, capped)
        emit({"stage": "inventory", "status": "start", "model": "muse-spark-1.3",
              "concepts": [], "objects": []})
        prog("Inventorying objects with Muse Spark 1.3…")
        inventory_called = not bool(concepts)
        objects = list_objects(bf, af, concepts=concepts)
        inventory_concepts = [o["concept"] for o in objects if isinstance(o, dict) and o.get("concept")]
        concept_list = _dedupe_prepend(watchlist_list, inventory_concepts, cap=MAX_CONCEPTS)
        known = {_norm_concept(o.get("concept", "")) for o in objects if isinstance(o, dict)}
        extra_objects = [{"concept": c, "where_before": "", "where_after": "",
                          "suspected_change": "unchanged"}
                         for c in concept_list if _norm_concept(c) not in known]
        if extra_objects:
            objects = extra_objects + list(objects)
        emit({"stage": "inventory", "status": "done", "model": "muse-spark-1.3",
              "concepts": list(concept_list), "objects": objects})
        prog(f"Concepts ({len(concept_list)}): {', '.join(concept_list) or 'none'}")

        # 3b. Coverage (Spark part) — after inventory, before tracking.
        # Final score also needs post-tracking anchor overlap, so the
        # "done" event is emitted after tracking (see 4c).
        emit({"stage": "coverage", "status": "start", "model": "muse-spark-1.3"})
        prog("Checking walkthrough coverage with Muse Spark 1.3…")
        coverage_spark = {"score": 1.0, "missing_in_after": [], "missing_in_before": [], "retake_tip": ""}
        coverage_error = None
        counters["coverage_calls"] += 1
        try:
            cov = check_coverage(bf, af) or {}
            try:
                spark_score_tmp = float(cov.get("score", 1.0) or 0.0)
            except Exception:
                spark_score_tmp = 1.0
            coverage_spark = {"score": max(0.0, min(1.0, spark_score_tmp)),
                              "missing_in_after": [str(a) for a in (cov.get("missing_in_after") or [])],
                              "missing_in_before": [str(a) for a in (cov.get("missing_in_before") or [])],
                              "retake_tip": str(cov.get("retake_tip", "") or "")}
        except Exception as e:
            coverage_error = str(e)
            log.exception("coverage Spark call failed; continuing with anchor overlap only")

        # 4. SAM tracking (concurrent; progress as each future completes)
        prog("Tracking concepts with SAM 3.1 (concurrent)…")
        tasks = [(c, "before", before_norm) for c in concept_list] + [(c, "after", after_norm) for c in concept_list]
        emit({"stage": "track", "status": "start", "total": len(tasks)})
        before_tracks, after_tracks = {}, {}
        sam_requests, sam_frames = 0, 0

        def _job(t):
            concept, side, path = t
            try:
                return concept, side, track_concept(path, concept)
            except Exception as e:
                log.exception("SAM failed for %s/%s", concept, side); return concept, side, {}

        done_count = 0
        if tasks:
            with ThreadPoolExecutor(max_workers=6) as ex:
                futures = {ex.submit(_job, t): t for t in tasks}
                for fut in as_completed(futures):
                    concept, side, parsed = fut.result()
                    parsed = parsed or {}
                    (before_tracks if side == "before" else after_tracks)[concept] = {
                        int(k): v for k, v in parsed.items()}
                    done_count += 1; sam_requests += 1
                    try:
                        n_frames = len(parsed)
                        n_dets = sum(len(v or []) for v in parsed.values())
                    except Exception:
                        n_frames, n_dets = 0, 0
                    sam_frames += n_frames
                    emit({"stage": "track", "status": "progress", "concept": concept,
                          "video": side, "frames": int(n_frames), "detections": int(n_dets),
                          "done": done_count, "total": len(tasks)})
        emit({"stage": "track", "status": "done"})
        summaries = diffmod.summarize_tracks(before_tracks, after_tracks)

        # 4b. Vocabulary repair
        repair = _run_vocab_repair(summaries, watchlist_list, before_norm, after_norm,
                                   bf, af, prog, emit=emit,
                                   before_tracks=before_tracks, after_tracks=after_tracks)
        aliases = repair.get("aliases", {}) or {}
        sam_requests += int(repair.get("sam_requests", 0) or 0)
        sam_frames += int(repair.get("sam_frames", 0) or 0)

        original_summaries = copy.deepcopy(summaries)

        # 5. SAM image re-checks (emits its own start/progress/done events)
        recheck_calls = _run_image_rechecks(summaries, bf, af, prog, emit=emit)
        rechecks_count = sum(1 for s in summaries.values() if (s.get("recheck") or {}).get("checked"))

        # 4c. Coverage (final) — deterministic anchor overlap after tracking.
        coverage = {"score": 1.0, "ok": True, "anchor_overlap": 1.0,
                    "spark_score": float(coverage_spark.get("score", 1.0) or 0.0),
                    "shared_anchors": [], "only_before": [], "only_after": [],
                    "missing_in_after": list(coverage_spark.get("missing_in_after") or []),
                    "missing_in_before": list(coverage_spark.get("missing_in_before") or []),
                    "retake_tip": str(coverage_spark.get("retake_tip", "") or "")}
        try:
            shared, only_b, only_a = [], [], []
            for concept, s in (summaries or {}).items():
                if not diffmod.is_anchor(concept):
                    continue
                b_present = bool((s.get("before") or {}).get("present"))
                a_present = bool((s.get("after") or {}).get("present"))
                if b_present and a_present:
                    shared.append(concept)
                elif b_present:
                    only_b.append(concept)
                elif a_present:
                    only_a.append(concept)
            either = len(shared) + len(only_b) + len(only_a)
            anchor_overlap = (len(shared) / either) if either else 1.0
            spark_score = float(coverage_spark.get("score", 1.0) or 0.0)
            final_score = min(spark_score, float(anchor_overlap))
            coverage = {"score": float(final_score), "ok": bool(final_score >= 0.6),
                        "anchor_overlap": float(anchor_overlap), "spark_score": float(spark_score),
                        "shared_anchors": shared, "only_before": only_b, "only_after": only_a,
                        "missing_in_after": list(coverage_spark.get("missing_in_after") or []),
                        "missing_in_before": list(coverage_spark.get("missing_in_before") or []),
                        "retake_tip": str(coverage_spark.get("retake_tip", "") or "")}
        except Exception as e:
            log.exception("coverage anchor computation failed; continuing")
            if coverage_error is None:
                coverage_error = str(e)
        cov_done = {"stage": "coverage", "status": "done", "score": float(coverage.get("score", 0) or 0),
                    "ok": bool(coverage.get("ok")), "retake_tip": str(coverage.get("retake_tip", "") or "")}
        if coverage_error:
            cov_done["error"] = coverage_error
        emit(cov_done)
        prog(f"Coverage: score={coverage['score']:.2f} ok={coverage['ok']} "
             f"(spark={coverage['spark_score']:.2f}, anchors={coverage['anchor_overlap']:.2f})")

        # 5b. Floor clutter check (grounded) — after tracking.
        floor_check = {"items": [], "dropped": []}
        floor_error = None
        emit({"stage": "floor", "status": "start", "model": "muse-spark-1.3"})
        prog("Checking the floor and walking path for clutter…")
        try:
            counters["floor_spark_calls"] += 1
            try:
                proposals = propose_floor_items(af) or []
            except Exception as e:
                log.warning("floor proposal failed: %s", e)
                proposals = []
                floor_error = str(e)
            proposals = [p for p in proposals if isinstance(p, dict) and p.get("concept")][:6]

            # Frame heights for the floor (y2) test, per AFTER keyframe.
            kf_heights: list[float] = []
            for kf in (af or []):
                hgt = 0.0
                try:
                    img_tmp = cv2.imread(str(kf))
                    if img_tmp is not None:
                        hgt = float(img_tmp.shape[0])
                except Exception:
                    hgt = 0.0
                kf_heights.append(hgt)

            def _floor_job(item):
                concept = " ".join(str(item.get("concept", "") or "").strip().lower().split())
                where = str(item.get("where", "") or "")
                best = None  # (c, box, kf_index, frame_h)
                calls = 0
                for idx, kf in enumerate(af or []):
                    calls += 1
                    try:
                        parsed = segment_image(kf, concept)
                    except Exception as e:
                        log.warning("floor SAM check failed for %s on %s: %s", concept, kf, e)
                        continue
                    for det in _all_detections(parsed):
                        frame_h = float(det.get("h") or 0) or (kf_heights[idx] if idx < len(kf_heights) else 0.0)
                        if frame_h <= 0:
                            continue
                        box = det["box"]
                        if float(det["c"]) >= FLOOR_MIN_CONF and float(box[3]) >= FLOOR_BOTTOM_FRAC * frame_h:
                            if best is None or float(det["c"]) > float(best[0]):
                                best = (float(det["c"]), [int(round(v)) for v in box], idx, frame_h)
                return {"concept": concept, "where": where, "_best": best, "_calls": calls}

            verified: list[dict] = []
            if proposals:
                with ThreadPoolExecutor(max_workers=6) as ex:
                    futures = [ex.submit(_floor_job, p) for p in proposals]
                    for fut in as_completed(futures):
                        try:
                            verified.append(fut.result())
                        except Exception:
                            log.exception("floor verification job failed")
            # Keep proposal order for deterministic output.
            order = {_norm_concept(p.get("concept", "")): i for i, p in enumerate(proposals)}
            verified.sort(key=lambda v: order.get(_norm_concept(v.get("concept", "")), 99))
            for v in verified:
                counters["floor_checks"] += int(v.pop("_calls", 0) or 0)
                best = v.pop("_best", None)
                concept = v["concept"]; where = v["where"]
                if best is None:
                    floor_check["dropped"].append({
                        "concept": concept,
                        "reason": (f"SAM did not confirm '{concept}' on the floor in the AFTER keyframes "
                                   f"(need c >= {FLOOR_MIN_CONF:.2f} and box bottom in the lower floor area)")})
                    continue
                c_val, box, kf_idx, _fh = best
                # Save a redacted, annotated image for this kept item.
                img_url = None
                try:
                    frame = cv2.imread(str((af or [])[kf_idx])) if af and 0 <= kf_idx < len(af) else None
                    if frame is None and af:
                        frame = cv2.imread(str(af[0]))
                    if frame is not None:
                        frame = _redact_frame_arr(frame)
                        fname = f"floor_{_safe(concept)}.jpg"
                        cv2.imwrite(str(out / fname), draw_box(frame, box, f"{concept}: on the floor", BRICK_RED))
                        img_url = _url(fname)
                except Exception as e:
                    log.warning("floor annotated image failed for %s: %s", concept, e)
                floor_check["items"].append({"concept": concept, "where": where,
                                             "c": float(c_val), "image": img_url,
                                             "box": box, "keyframe_index": int(kf_idx)})
            prog(f"Floor check: {len(floor_check['items'])} kept, {len(floor_check['dropped'])} dropped.")

"""Muse Voice Transcribe (muse-voice-transcribe-1.0) support.

Converts any audio/video input to 16 kHz mono WAV with ffmpeg, sends it to
POST https://api.meta.ai/v1/asr/transcribe, and (via Muse Spark 1.3) extracts
a caregiver watchlist of objects that must stay in place / within reach.

With ``diarize=True``, transcription uses request mode ``DIARIZATION`` and
``Accept: text/plain``; the response is one turn per line as ``A:text`` /
``B:text`` (``unknown:text`` if unattributed), parsed into ``turns``.
:meth:`extract_intent` additionally pulls out changes a speaker says were
made ON PURPOSE ("Dad's say").
"""

import logging
import os

import subprocess
import tempfile
import time
from pathlib import Path

import requests

from . import meta_api

log = logging.getLogger("splatdiff.voice")

MODEL = "muse-voice-transcribe-1.0"
ASR_URL = "https://api.meta.ai/v1/asr/transcribe"
# $0.18 per audio hour.
COST_PER_AUDIO_HOUR_USD = 0.18
MAX_SECONDS = 600  # API max: 10 min
MAX_BYTES = 32 * 1024 * 1024  # API max: 32 MB

_NO_STREAM_PHRASES = (
    "does not contain any stream",
    "output file does not contain any stream",
    "matches no streams",
    "no stream",

_TURN_RE = re.compile(r"^\s*([^:\n]{1,32}?)\s*:\s*(.+)$")

def _ffmpeg_exe() -> str:
    from imageio_ffmpeg import get_ffmpeg_exe

    return get_ffmpeg_exe()


def _probe_has_audio_stream(src: Path) -> bool | None:
    """Return True/False if ffmpeg's banner clearly shows an audio stream, else None."""
    try:
        proc = subprocess.run(
            [_ffmpeg_exe(), "-hide_banner", "-i", str(src)],
            capture_output=True,
            text=True,
            timeout=60,
        )
    except Exception as exc:  # noqa: BLE001
        log.warning("audio probe failed for %s: %s", src, exc)
        return None
    banner = (proc.stderr or "") + (proc.stdout or "")
    if ": Audio:" in banner:
        return True
    # ffmpeg -i without an output exits non-zero but prints stream info; if it
    # printed any Stream lines and none was Audio, there is no audio stream.
    if "Stream #" in banner:
        return False
    return None


def _convert_to_wav(src: Path) -> Path | None:
    """Convert to 16 kHz mono s16 WAV. Returns None if there is no audio stream."""
    tmp_dir = Path("cache") / "voice_tmp"
    tmp_dir.mkdir(parents=True, exist_ok=True)
    fd, tmp_name = tempfile.mkstemp(prefix="splatdiff_voice_", suffix=".wav", dir=str(tmp_dir))
    os.close(fd)
    out = Path(tmp_name)
    cmd = [
        _ffmpeg_exe(),
        "-y",
        "-v", "error",
        "-i", str(src),
        "-vn",
        "-ac", "1",
        "-ar", "16000",
        "-sample_fmt", "s16",
        "-t", str(MAX_SECONDS),
        str(out),
    ]
    try:
        proc = subprocess.run(cmd, capture_output=True, text=True, timeout=600)
    except Exception as exc:  # noqa: BLE001
        out.unlink(missing_ok=True)
        raise RuntimeError(f"ffmpeg audio conversion failed: {exc}") from exc
    if proc.returncode != 0:
        stderr = (proc.stderr or "").strip()
        out.unlink(missing_ok=True)
        if any(phrase in stderr.lower() for phrase in _NO_STREAM_PHRASES):
            log.info("no audio stream in %s", src)
            return None
        raise RuntimeError(f"ffmpeg audio conversion failed: {stderr[:1000] or 'unknown error'}")
    if not out.is_file() or out.stat().st_size <= 44:
        out.unlink(missing_ok=True)
        log.info("no usable audio produced from %s", src)
        return None
    return out


def _wav_duration_ms(wav_path: Path) -> int:
    try:
        size = wav_path.stat().st_size
    except Exception:  # noqa: BLE001
        return 0
    # 16000 samples/s * 1 channel * 2 bytes/sample = 32000 bytes/s (plus header).
    payload = max(0, size - 44)
    return int(payload / 32000.0 * 1000)


def _parse_turns(text: str) -> list[dict]:
    """Parse DIARIZATION text/plain output: one turn per line, 'A:text'."""
    turns: list[dict] = []
    for line in str(text or "").splitlines():
        line = line.strip()
        if not line:
            continue
        m = _TURN_RE.match(line)
        if m:
            speaker = m.group(1).strip() or "unknown"
            turn_text = m.group(2).strip()
        else:
            speaker, turn_text = "unknown", line
        if turn_text:
            turns.append({"speaker": speaker, "text": turn_text})
    return turns


def transcribe(path, diarize: bool = False) -> dict:
    """Transcribe an audio/video file.

    Returns {'transcript': str, 'duration_ms': int, 'latency_s': float}.
    With ``diarize=True`` the ASR request uses mode ``DIARIZATION`` and
    ``Accept: text/plain``, and the result also carries
    ``'turns': [{'speaker', 'text'}]`` parsed from the one-turn-per-line
    response. A file with no audio stream yields an empty transcript
    (no API call).
    """
    start = time.time()
    src = Path(path)
    if not src.is_file():
        raise FileNotFoundError(f"audio/video file not found: {src}")

    def _empty() -> dict:
        out = {"transcript": "", "duration_ms": 0, "latency_s": round(time.time() - start, 3)}
        if diarize:
            out["turns"] = []
        return out

    if _probe_has_audio_stream(src) is False:
        return _empty()

    wav_path = _convert_to_wav(src)
    if wav_path is None:
        return _empty()

    try:
        wav_bytes = wav_path.read_bytes()
        if len(wav_bytes) > MAX_BYTES:  # -t 600 should prevent this; be defensive.
            wav_bytes = wav_bytes[:MAX_BYTES]
        fallback_duration_ms = _wav_duration_ms(wav_path)

        request_json = json.dumps(
            {"model": MODEL, "audioEncoding": "WAV",
             "mode": "DIARIZATION" if diarize else "PUSH_TO_TALK"}
        )
        headers = {"Authorization": f"Bearer {_api_key()}",
                   "Accept": "text/plain" if diarize else "application/json"}

        last_err: Exception | None = None
        api_start = time.time()
        for attempt in range(3):
            api_start = time.time()
            try:
                resp = requests.post(
                    ASR_URL,
                    headers=headers,
                    files={
                        "request": (None, request_json, "application/json"),
                        "audio": ("note.wav", wav_bytes, "audio/wav"),
                    },
                    timeout=300,
                )
                if resp.status_code == 400:
                    print(f"Voice Transcribe 400 error body: {resp.text[:4000]}")
                    resp.raise_for_status()
                if resp.status_code in (429, 500, 502, 503, 504):
                    raise requests.HTTPError(
                        f"retryable {resp.status_code}: {resp.text[:1000]}", response=resp
                    )
                resp.raise_for_status()
                api_latency = time.time() - api_start

                if diarize:
                    turns = _parse_turns(resp.text or "")
                    transcript = "\n".join(f"{t['speaker']}: {t['text']}" for t in turns)
                    duration_ms = fallback_duration_ms
                else:
                    data = resp.json()
                    transcript = data.get("transcript") or ""
                    if not isinstance(transcript, str):
                        transcript = str(transcript)
                    try:
                        duration_ms = int(data.get("audioDurationMs") or fallback_duration_ms)
                    except Exception:  # noqa: BLE001
                        duration_ms = fallback_duration_ms

                est_usd = COST_PER_AUDIO_HOUR_USD * (duration_ms / 1000.0 / 3600.0)
                _log_cost(MODEL, "transcribe", est_usd, api_latency)

            except requests.HTTPError as exc:

def _clean_items(raw_items, limit: int = 8) -> list[str]:

def _room_slug(room) -> str:

def _parse_date(value, fallback_path: Optional[Path] = None):
    """Best-effort parse of a date/run_id into an aware datetime."""
    if isinstance(value, str) and value.strip():
        txt = value.strip()
        for fmt in ("%Y%m%d_%H%M%S", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S", "%Y-%m-%d"):
            try:
                return datetime.strptime(txt[:19] if fmt != "%Y%m%d_%H%M%S" else txt[:15], fmt).astimezone()
            except Exception:
                pass
        try:
            dt = datetime.fromisoformat(txt.replace("Z", "+00:00"))
            return dt if dt.tzinfo else dt.astimezone()
        except Exception:
            pass
        # run_id embedded at start
        m = re.match(r"^(\d{8}_\d{6})", txt)
        if m:
            try:
                return datetime.strptime(m.group(1), "%Y%m%d_%H%M%S").astimezone()
            except Exception:
                pass
    if fallback_path is not None:
        try:
            return datetime.fromtimestamp(fallback_path.stat().st_mtime).astimezone()
        except Exception:
            pass
    return None


def _result_datetime(result: dict, path: Optional[Path] = None):

def _watchlist_entries(result: dict):
    """Return list of watchlist item dicts from any plausible result shape."""
    if not isinstance(result, dict):
        return []
    candidates = []
    for key in ("watchlist_results", "watchlist_status", "watched_items", "watchlist_items"):
        val = result.get(key)
        if isinstance(val, list):
            candidates.extend([v for v in val if isinstance(v, dict)])
        elif isinstance(val, dict):
            candidates.extend([v for v in val.values() if isinstance(v, dict)])
    val = result.get("watchlist")
    if isinstance(val, list):
        for v in val:
            if isinstance(v, dict):
                candidates.append(v)
    elif isinstance(val, dict):
        for k in ("items", "results", "entries"):
            sub = val.get(k)
            if isinstance(sub, list):
                candidates.extend([v for v in sub if isinstance(v, dict)])
    return candidates


def _watchlist_names(result: dict):
    names = []
    if not isinstance(result, dict):
        return names
    for entry in _watchlist_entries(result):
        name = entry.get("concept") or entry.get("name") or entry.get("item")
        if name:
            names.append(str(name))
    val = result.get("watchlist")
    if isinstance(val, list):
        for v in val:
            if isinstance(v, str) and v.strip():
                names.append(v.strip())
    elif isinstance(val, dict):
        for k in ("items", "concepts", "names"):
            sub = val.get(k)
            if isinstance(sub, list):
                names.extend([str(v) for v in sub if isinstance(v, str) and v.strip()])
    # de-dupe, preserve order
    seen, out = set(), []
    for n in names:
        key = n.lower()
        if key not in seen:
            seen.add(key)
            out.append(n)
    return out

def _next_step(change, risk=None, intentional=False) -> str:
    """Change-specific plain next step for a family-feed item."""
    if intentional:
        return "Dad says this was on purpose — consider making it the new setup."
    c = str(change or "").strip().lower()
    if c == "moved":
        return "Ask Dad if he moved it on purpose; if not, someone there puts it back with him."
    if c in ("removed", "missing"):
        return "Ask Dad where it is; put it back where he expects it."
    if c in ("added", "appeared", "on_floor"):
        return "Move it out of the walking path, with Dad."
    if c == "not_seen":
        return "Check it in person next visit."
    return "Have a look at this on the next visit."

def _feed_status(result: dict) -> str:
    coverage = result.get("coverage") if isinstance(result, dict) else None
    if isinstance(coverage, dict) and coverage.get("ok") is False:
        return "coverage_gap"
    for claim in _kept_claims(result):
        if str(claim.get("risk") or "").strip().lower() == "high":
            return "needs_attention"
    for entry in _watchlist_entries(result):
        status = str(entry.get("status") or entry.get("state") or "").strip().lower()
        if status in ("moved", "missing", "not_seen"):
            return "needs_attention"
    return "all_clear"


def _feed_entry_from_result(result: dict, path: Optional[Path] = None) -> dict:
    room = str(result.get("room") or "Bedroom") if isinstance(result, dict) else "Bedroom"
    run_id = str(result.get("run_id") or (path.parent.name if path else uuid.uuid4().hex[:12]))
    dt = _result_datetime(result, path)
    status = _feed_status(result)
    items = []
    # Kept claims first: items are deduped by concept, and the kept claim
    # for a concept is preferred over its watchlist row.
    seen_concepts: set = set()
    for claim in _kept_claims(result):
        concept = str(claim.get("concept") or claim.get("name") or "item")
        key = concept.strip().lower()
        if key in seen_concepts:
            continue
        seen_concepts.add(key)
        change = str(claim.get("change") or claim.get("label") or "")
        risk = claim.get("risk")
        is_intentional = bool(claim.get("intentional"))
        items.append({
            "concept": concept,
            "change": change,
            "risk": risk,
            "next_step": _next_step(change, risk, intentional=is_intentional),
        })
    # Surface concerning watchlist items not already represented.
    for entry in _watchlist_entries(result):
        st = str(entry.get("status") or entry.get("state") or "").strip().lower()
        if st in ("moved", "missing", "not_seen"):
            concept = str(entry.get("concept") or entry.get("name") or entry.get("item") or "watched item")
            if concept.strip().lower() not in seen_concepts:
                items.append({"concept": concept, "change": st, "risk": entry.get("risk"),
                              "next_step": _next_step(st, entry.get("risk"))})
                seen_concepts.add(concept.strip().lower())
    if status == "coverage_gap":
        headline = f"{room} — coverage gap: the walkthroughs didn't cover the same corners"
    elif status == "needs_attention":
        first = items[0] if items else None
        if first:
            headline = f"{room} — needs attention: {first['concept']} {first['change']}".strip()
        else:
            headline = f"{room} — needs attention"
    else:
        if _watchlist_names(result) or _watchlist_entries(result):
            headline = f"{room} — all watched items seen where they were"
        else:
            headline = f"{room} — all clear"
    return {
        "id": run_id,
        "ts": dt.isoformat(),
        "room": room,
        "filmed_by": result.get("filmed_by") if isinstance(result, dict) else None,
        "status": status,
        "headline": headline,
        "items": items,
        "run_id": run_id,
        "trend": _result_trend_text(result),
    }

def _feed_load():
    with FEED_LOCK:
        FEED.clear()
        if FEED_PATH.is_file():
            try:
                for line in FEED_PATH.read_text(encoding="utf-8").splitlines():
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        obj = json.loads(line)
                        if isinstance(obj, dict):
                            FEED.append(obj)
                    except Exception:
                        continue
                log.info("feed loaded %d entries from %s", len(FEED), FEED_PATH)
                return
            except Exception as exc:  # noqa: BLE001
                log.warning("could not load feed, rebuilding: %s", exc)
                FEED.clear()
        # Missing (or unreadable): rebuild from outputs/*/result.json.
        FEED.extend(_feed_rebuild_from_outputs())
        _feed_persist_all_locked()
        log.info("feed rebuilt %d entries from outputs/", len(FEED))

def _collect_video_candidates(result, after_only=False):
    """Collect .mp4 paths referenced by a result (defensive, recursive)."""
    found = []

    def walk(obj, key_hint=""):
        if isinstance(obj, dict):
            for k, v in obj.items():
                walk(v, str(k).lower())
        elif isinstance(obj, list):
            for v in obj:
                walk(v, key_hint)
        elif isinstance(obj, str):
            low = obj.lower()
            if low.endswith(".mp4"):
                if after_only and key_hint and "before" in key_hint:
                    return
                found.append(obj)

    walk(result)
    # Also any mp4 sitting in the run's outputs folder.
    if isinstance(result, dict) and result.get("run_id"):
        out_dir = ROOT / "outputs" / str(result["run_id"])
        if out_dir.is_dir():
            for p in sorted(out_dir.glob("*.mp4")):
                found.append(str(p))
    seen, out = set(), []
    for f in found:
        if f not in seen:
            seen.add(f)
            out.append(f)
    return out

        return None
    items = [s.strip() for s in str(value).split(",") if s.strip()]
    return items or None


def _merge_watchlist(existing, extra):
    """Merge two watchlists, deduping case-insensitively, preserving order."""
    merged = []
    seen = set()
    for source in (existing or [], extra or []):
        for item in source:
            if not isinstance(item, str):
                continue
            cleaned = item.strip()
            if not cleaned:
                continue
            key = cleaned.lower()
            if key in seen:
                continue
            seen.add(key)
            merged.append(cleaned)
    return merged


def _json_safe(obj):
    return json.loads(json.dumps(obj, default=str))


def _new_state():
    return {"status": "running", "events": [], "cond": threading.Condition(), "result": None}


def _append_event(state, event):
    with state["cond"]:
        state["events"].append(event)
        state["cond"].notify_all()


def _delete_run_media(run_id, before, after, result, keep_media, protected=()):
    """Delete uploaded raw videos + the run's normalized videos (unless kept).

    Annotated images live in outputs/ and are never touched, so history
    (which references outputs/ images) keeps working after deletion.
    """
    if keep_media:
        log.info("run %s keep_media=1: uploaded/normalized videos kept", run_id)
        return
    protected_resolved = set()
    for item in protected or ():
        try:
            protected_resolved.add(str(Path(str(item)).resolve()))
        except Exception:
            protected_resolved.add(str(item))
    deleted = []
    # Normalized (and any other result-referenced) videos outside outputs/.
    if isinstance(result, dict):
        for cand in _collect_video_candidates(result):
            p = _resolve_local_path(cand)
            if p is None:
                continue
            try:
                resolved = str(p.resolve())
                if resolved in protected_resolved:
                    continue
                if (ROOT / "outputs") in p.resolve().parents:
                    continue
                if (ROOT / "data") in p.resolve().parents:
                    continue
                p.unlink()
                deleted.append(str(p))
            except Exception as exc:  # noqa: BLE001
                log.warning("run %s could not delete media %s: %s", run_id, cand, exc)
    # Uploaded raw videos for this server run.
    upload_dir = ROOT / "cache" / "uploads" / run_id
    try:
        if upload_dir.is_dir():
            shutil.rmtree(upload_dir, ignore_errors=True)
            deleted.append(str(upload_dir))
    except Exception as exc:  # noqa: BLE001
        log.warning("run %s could not delete uploads %s: %s", run_id, upload_dir, exc)
    # Fall back to deleting the exact input files if they live under uploads/.
    for raw in (before, after):
        try:
            p = Path(str(raw))
            if p.is_file() and (ROOT / "cache" / "uploads") in p.resolve().parents and str(p.resolve()) not in protected_resolved:
                p.unlink()
                deleted.append(str(p))
        except Exception:
            pass
    log.info("run %s media deleted after run (keep_media=0): %s", run_id, deleted or "nothing to delete")


def _run_pipeline(run_id: str, before: str, after: str, concepts, watchlist, narration: bool = False,
                  room: str = "Bedroom", filmed_by=None, keep_media: bool = False, protected_paths=()):
    state = RUNS[run_id]
    run_started = time.time()
    log.info("run %s start before=%s after=%s concepts=%s watchlist=%s narration=%s room=%s filmed_by=%s keep_media=%s",
             run_id, before, after, concepts, watchlist, narration, room, filmed_by, keep_media)

    def cb(event):
        try:
            ev = _json_safe(event) if isinstance(event, dict) else {"stage": "log", "status": "info", "message": str(event)}
        except Exception:
            ev = {"stage": "log", "status": "info", "message": str(event)}
        _append_event(state, ev)

    voice_info = None
    effective_watchlist = watchlist
    intentional_for_pipeline = None
    if narration:
        try:
            voice_started = time.time()
            transcribed = voice_mod.transcribe(before, diarize=True)
            transcript = transcribed.get("transcript", "") if isinstance(transcribed, dict) else ""
            turns = transcribed.get("turns", []) if isinstance(transcribed, dict) else []
            if not isinstance(turns, list):
                turns = []
            intent = voice_mod.extract_intent(turns if turns else transcript)
            if not isinstance(intent, dict):
                intent = {"items": [], "intentional": []}
            items = intent.get("items") or []
            intentional = intent.get("intentional") or []
            effective_watchlist = _merge_watchlist(watchlist, items) or None
            intentional_for_pipeline = intentional or None
            voice_info = {"transcript": transcript, "items": items,
                          "intentional": intentional, "turns": turns}
            _append_event(state, {
                "stage": "voice",
                "status": "done",
                "model": "muse-voice-transcribe-1.0",
                "transcript": transcript,
                "items": items,
                "intentional": intentional,
                "turns": turns,
                "elapsed_s": round(time.time() - run_started, 3),
                "cost_usd": 0.0,
            })
            log.info("run %s voice note done items=%s intentional=%s (%.2fs)",
                     run_id, items, intentional, time.time() - voice_started)
        except Exception as exc:  # noqa: BLE001
            log.exception("run %s narration failed, continuing without it: %s", run_id, exc)
            _append_event(state, {"stage": "voice", "status": "error", "message": str(exc)})

    try:
        try:
            result = pipeline.run(before, after, concepts=concepts or None, watchlist=effective_watchlist or None,
                                  on_event=cb, room=room, filmed_by=filmed_by,
                                  intentional=intentional_for_pipeline)
        except TypeError as exc:
            # Older pipeline without room/filmed_by/intentional kwargs: retry without them.
            if ("room" not in str(exc) and "filmed_by" not in str(exc)
                    and "intentional" not in str(exc) and "unexpected keyword" not in str(exc)):
                raise
            log.warning("run %s pipeline rejected room/filmed_by/intentional kwargs, retrying without: %s", run_id, exc)
            result = pipeline.run(before, after, concepts=concepts or None, watchlist=effective_watchlist or None, on_event=cb)
            if isinstance(result, dict):
                result.setdefault("room", room)
                result.setdefault("filmed_by", filmed_by)
        if result is None:
            raise RuntimeError("pipeline returned no result")
        if voice_info is not None and isinstance(result, dict):
            models = result.get("models")
            if not isinstance(models, list):
                models = [models] if models else []
                result["models"] = models
            if "muse-voice-transcribe-1.0" not in models:
                models.append("muse-voice-transcribe-1.0")
            result["narration"] = {"transcript": voice_info["transcript"], "items": voice_info["items"],
                                   "intentional": voice_info.get("intentional", []),
                                   "turns": voice_info.get("turns", [])}
        if isinstance(result, dict):
            result.setdefault("room", room)
            if filmed_by is not None:
                result.setdefault("filmed_by", filmed_by)
        safe_result = _json_safe(result)
        # Remember where the AFTER videos are, for POST /api/baseline.
        if isinstance(safe_result, dict) and safe_result.get("run_id"):
            with RUN_MEDIA_LOCK:
                RUN_MEDIA[str(safe_result["run_id"])] = {
                    "after_original": str(after),
                    "before_original": str(before),
                    "after_candidates": _collect_video_candidates(safe_result, after_only=False),
                    "result": safe_result,
                }
        # Family feed: append when a run finishes (text only).
        try:
            if isinstance(safe_result, dict):
                result_path = ROOT / "outputs" / str(safe_result.get("run_id") or "") / "result.json"
                _feed_append(_feed_entry_from_result(safe_result, result_path if result_path.is_file() else None))
        except Exception as exc:  # noqa: BLE001
            log.warning("run %s feed append failed: %s", run_id, exc)
        # Delete raw/normalized media by default; outputs/ stays for history.
        _delete_run_media(run_id, before, after, safe_result if isinstance(safe_result, dict) else {}, keep_media, protected=protected_paths)
        # Append full-result event BEFORE marking status done, so SSE clients get it.
        final_event = {"stage": "result", "status": "done", "result": safe_result}
        with state["cond"]:
            state["events"].append(final_event)
            state["cond"].notify_all()
            state["result"] = safe_result
            state["status"] = "done"
        log.info("run %s done pipeline_run_id=%s", run_id, safe_result.get("run_id") if isinstance(safe_result, dict) else "?")
    except Exception as exc:  # noqa: BLE001
        log.exception("run %s failed: %s", run_id, exc)
        _delete_run_media(run_id, before, after, {}, keep_media, protected=protected_paths)
        err_event = {"stage": "error", "status": "error", "message": str(exc)}
        with state["cond"]:
            # Avoid duplicate error events if pipeline already emitted one.
            if not any(isinstance(e, dict) and e.get("stage") == "error" for e in state["events"]):
                state["events"].append(err_event)
            state["status"] = "error"
            state["cond"].notify_all()

async def api_create_run(
    before: Optional[UploadFile] = File(None),
    after: Optional[UploadFile] = File(None),
    sample: Optional[str] = Form(None),
    watchlist: Optional[str] = Form(None),
    concepts: Optional[str] = Form(None),
    narration: Optional[str] = Form(None),
    room: Optional[str] = Form(None),
    filmed_by: Optional[str] = Form(None),
    baseline: Optional[str] = Form(None),
    keep_media: Optional[str] = Form(None),
):
    run_id = uuid.uuid4().hex[:12]
    concepts_list = _split_csv(concepts)
    watchlist_list = _split_csv(watchlist)
    narration_enabled = bool(narration and str(narration).strip() == "1")
    room_name = (str(room).strip() if room and str(room).strip() else "Bedroom")
    slug = _room_slug(room_name)

    use_baseline = bool(baseline and str(baseline).strip() == "1")
    keep = bool(keep_media and str(keep_media).strip() == "1")
    is_sample = bool(sample and str(sample).strip() == "1")

    protected = []
    baseline_path = None
    if use_baseline:
        candidate = BASELINES_DIR / f"{slug}.mp4"
        if not candidate.is_file():
            return JSONResponse({"error": f"no baseline saved for room '{room_name}'"}, status_code=400)
        baseline_path = candidate
        protected.append(str(candidate))

    if is_sample and not use_baseline:
        before_path = ROOT / "data" / "sample_before.mp4"
        after_path = ROOT / "data" / "sample_after.mp4"
        protected.extend([str(before_path), str(after_path)])
    else:
        # Baseline supplies BEFORE; sample can supply AFTER if none uploaded.
        if use_baseline and after is None and is_sample:
            after_path = ROOT / "data" / "sample_after.mp4"
            protected.append(str(after_path))
            before_path = baseline_path
        elif use_baseline:
            if after is None:
                return JSONResponse({"error": "after file is required when using a baseline"}, status_code=400)
            before_path = baseline_path
            after_path = None  # saved below
        else:
            if before is None or after is None:
                return JSONResponse({"error": "before and after files are required unless sample=1"}, status_code=400)

            after_path = None

        if before_path is None or after_path is None:
            upload_dir = ROOT / "cache" / "uploads" / run_id
            upload_dir.mkdir(parents=True, exist_ok=True)

            async def _save(upload: UploadFile, stem: str) -> Path:

    state = _new_state()
    with RUNS_LOCK:
        RUNS[run_id] = state

    thread = threading.Thread(
        target=_run_pipeline,
        args=(run_id, str(before_path), str(after_path), concepts_list, watchlist_list, narration_enabled),
        kwargs={"room": room_name, "filmed_by": filmed_by_val, "keep_media": keep, "protected_paths": tuple(protected)},

    )
    thread.start()
    log.info("run %s queued thread=%s room=%s baseline=%s keep_media=%s sample=%s", run_id, thread.name, room_name, use_baseline, keep, is_sample)
    return {"run_id": run_id}


@app.get("/api/runs/{run_id}/events")
async def api_run_events(run_id: str):
    with RUNS_LOCK:
        state = RUNS.get(run_id)
    if state is None:
        return JSONResponse({"error": "run not found"}, status_code=404)

    async def event_stream():
        idx = 0
        last_ping = time.monotonic()
        while True:
            # Poll the shared list without blocking the event loop.
            with state["cond"]:
                pending = list(state["events"][idx:])
                status = state["status"]
            sent_terminal = False
            for ev in pending:
                idx += 1
                last_ping = time.monotonic()
                yield f"data: {json.dumps(ev, default=str)}\n\n"
                if isinstance(ev, dict) and ev.get("stage") in ("result", "error"):
                    sent_terminal = True
            if sent_terminal:
                return
            # Fallback: thread finished but no terminal result/error event was produced.
            with state["cond"]:
                drained = idx >= len(state["events"])
            if status in ("done", "error") and drained and pending == []:
                return
            now = time.monotonic()
            if now - last_ping >= 15:
                last_ping = now
                yield ": ping\n\n"
            await asyncio.sleep(0.25)

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )

@app.post("/api/baseline")
async def api_set_baseline(request: Request):
    try:
        body = await request.json()
    except Exception:
        return JSONResponse({"error": "JSON body required"}, status_code=400)
    if not isinstance(body, dict):
        return JSONResponse({"error": "JSON body required"}, status_code=400)
    pipeline_run_id = str(body.get("run_id") or "").strip()
    room_name = str(body.get("room") or "Bedroom").strip() or "Bedroom"
    slug = _room_slug(room_name)
    if not slug:
        return JSONResponse({"error": "invalid room name"}, status_code=400)
    if not pipeline_run_id or "/" in pipeline_run_id or "\\" in pipeline_run_id or ".." in pipeline_run_id:
        return JSONResponse({"error": "run not found"}, status_code=404)

    result = None
    result_path = ROOT / "outputs" / pipeline_run_id / "result.json"
    if result_path.is_file():
        try:
            result = json.loads(result_path.read_text(encoding="utf-8"))
        except Exception:
            result = None
    if not isinstance(result, dict):
        # Fall back to in-memory run result.
        with RUN_MEDIA_LOCK:
            info = RUN_MEDIA.get(pipeline_run_id)
        if info and isinstance(info.get("result"), dict):
            result = info["result"]
        else:
            return JSONResponse({"error": "run not found"}, status_code=404)

    candidates = []
    with RUN_MEDIA_LOCK:
        info = RUN_MEDIA.get(pipeline_run_id) or {}
        candidates.extend(info.get("after_candidates") or [])
        if info.get("after_original"):
            candidates.append(info["after_original"])
    candidates.extend(_collect_video_candidates(result))
    # Prefer AFTER-looking / normalized candidates, then any candidate.
    ordered = sorted(candidates, key=lambda c: (0 if "after" in Path(str(c)).name.lower() else 1, 0 if "normal" in str(c).lower() else 1))
    source = None
    for cand in ordered:
        p = _resolve_local_path(cand)
        if p is not None:
            # When the hint says "before", skip obvious before files if an after exists later.
            if "before" in p.name.lower() and any("after" in Path(str(c)).name.lower() for c in ordered):
                continue
            source = p
            break
    if source is None:
        return JSONResponse({"error": "Videos were deleted after the check; film a fresh baseline."}, status_code=409)

    try:
        BASELINES_DIR.mkdir(parents=True, exist_ok=True)
        dest = BASELINES_DIR / f"{slug}.mp4"
        shutil.copyfile(str(source), str(dest))
        meta = {"room": room_name, "date": _now_iso(), "from_run": pipeline_run_id, "watchlist": _watchlist_names(result)}
        (BASELINES_DIR / f"{slug}.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")
    except Exception as exc:  # noqa: BLE001
        log.exception("baseline copy failed: %s", exc)
        return JSONResponse({"error": str(exc)}, status_code=500)
    log.info("baseline saved room=%s slug=%s from_run=%s source=%s", room_name, slug, pipeline_run_id, source)
    return _json_safe(meta)

@app.get("/api/feed/events")
async def api_feed_events():
    async def feed_stream():
        with FEED_LOCK:
            snapshot_entries = list(reversed(FEED))
            idx = len(FEED)
        yield f"data: {json.dumps({'type': 'snapshot', 'entries': snapshot_entries, 'rooms': _compute_rooms()}, default=str)}\n\n"
        last_ping = time.monotonic()
        last_due = time.monotonic()
        while True:
            await asyncio.sleep(0.5)
            # Poll the feed list index every 0.5 s (lock makes this thread-safe).
            with FEED_LOCK:
                new_entries = list(FEED[idx:])
                idx = len(FEED)
            for entry in new_entries:
                last_ping = time.monotonic()
                yield f"data: {json.dumps({'type': 'entry', 'entry': entry}, default=str)}\n\n"
            now = time.monotonic()
            if now - last_due >= 60:
                last_due = now
                last_ping = now
                yield f"data: {json.dumps({'type': 'due', 'rooms': _compute_rooms()}, default=str)}\n\n"
                continue
            if now - last_ping >= 15:
                last_ping = now
                yield ": ping\n\n"

    return StreamingResponse(
        feed_stream(),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )

    start = (datetime.now() + timedelta(days=1)).replace(hour=10, minute=0, second=0, microsecond=0)
    dtstart = start.strftime("%Y%m%dT%H%M%S")
    dtstamp = datetime.now().strftime("%Y%m%dT%H%M%SZ")
    uid = f"{uuid.uuid4().hex}@secondlook"
    lines = [
        "BEGIN:VCALENDAR",
        "VERSION:2.0",
        "PRODID:-//Second Look//Reminder//EN",
        "BEGIN:VEVENT",
        f"UID:{uid}",
        f"DTSTAMP:{dtstamp}",
        f"DTSTART:{dtstart}",
        f"RRULE:FREQ=DAILY;INTERVAL={interval}",
        f"SUMMARY:Second Look: re-film Dad's {room_name}",
        "DESCRIPTION:Same route: door → bed → bathroom door. Lights on. 15–30 seconds.",
        "END:VEVENT",
        "END:VCALENDAR",
    ]
    body = "\r\n".join(lines) + "\r\n"
    return Response(
        content=body,
        media_type="text/calendar",
        headers={"Content-Disposition": 'attachment; filename="reminder.ics"'},
    )


@app.delete("/api/history/{run_id}")
async def api_delete_history(run_id: str):
    if not _RUN_ID_RE.match(run_id or ""):
        return JSONResponse({"error": "invalid run_id"}, status_code=400)
    target = ROOT / "outputs" / run_id
    if not target.is_dir():
        return JSONResponse({"error": "run not found"}, status_code=404)
    try:
        shutil.rmtree(target)
    except Exception as exc:  # noqa: BLE001
        log.exception("delete history failed: %s", exc)
        return JSONResponse({"error": str(exc)}, status_code=500)
    _feed_remove(run_id)
    with RUN_MEDIA_LOCK:
        RUN_MEDIA.pop(run_id, None)
    log.info("history deleted run_id=%s", run_id)
    return {"deleted": run_id}


@app.get("/api/history")
async def api_history():
    outputs_dir = ROOT / "outputs"
    entries = []
    if outputs_dir.is_dir():
        files = sorted(outputs_dir.glob("*/result.json"), key=lambda p: p.stat().st_mtime if p.exists() else 0, reverse=True)
        for path in files:
            try:
                result = json.loads(path.read_text(encoding="utf-8"))
            except Exception:
                continue
            try:
                if not isinstance(result, dict):
                    continue
                kept_claims = result.get("kept_claims") or result.get("claims") or []
                dropped_claims = result.get("dropped") or []
                thumb = None
                for claim in kept_claims:
                    if not isinstance(claim, dict):
                        continue
                    images = claim.get("images") or {}
                    if isinstance(images, dict) and (images.get("after") or images.get("before")):
                        thumb = images.get("after") or images.get("before")
                        break
                if not thumb:
                    keyframes = result.get("keyframes") or {}
                    if isinstance(keyframes, dict):
                        for side in ("after", "before"):
                            urls = keyframes.get(side) or []
                            if urls:
                                thumb = urls[0]
                                break
                entries.append(
                    {
                        "run_id": result.get("run_id") or path.parent.name,
                        "summary": result.get("summary") or "",
                        "kept": len(kept_claims),
                        "dropped": len(dropped_claims),
                        "cost_usd": result.get("cost_usd"),
                        "latency_s": result.get("latency_s"),
                        "thumb": thumb,
                        "room": result.get("room"),
                        "date": _result_datetime(result, path).isoformat(),
                    }
                )
            except Exception:
                continue
    return _json_safe(entries)


@app.get("/api/history/{run_id}")
async def api_history_one(run_id: str):
    # Prevent path traversal.
    if not run_id or "/" in run_id or "\\" in run_id or ".." in run_id:
        return JSONResponse({"error": "run not found"}, status_code=404)
    path = ROOT / "outputs" / run_id / "result.json"
    if not path.is_file():
        return JSONResponse({"error": "run not found"}, status_code=404)
    try:
        result = json.loads(path.read_text(encoding="utf-8"))
    except Exception as exc:
        return JSONResponse({"error": f"unreadable result: {exc}"}, status_code=500)
    return _json_safe(result)
